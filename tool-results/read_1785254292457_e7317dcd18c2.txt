    81→// Spreadsheet Preview Sub-component
    82→// Renders xlsx data as a scrollable table with sheet tabs
    83→// ============================================================
    84→function SpreadsheetPreview({ data, name }: {
    85→  data: {
    86→    sheetNames: string[];
    87→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
    88→  };
    89→  name: string;
    90→}) {
    91→  const [activeSheet, setActiveSheet] = useState(data.sheetNames[0] || '');
    92→  const sheetData = data.sheets[activeSheet];
    93→  const maxRows = 100;
    94→
    95→  if (!sheetData) {
    96→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    97→  }
    98→
    99→  return (
   100→    <div className="flex flex-col w-full">
   101→      {/* Sheet tabs */}
   102→      {data.sheetNames.length > 1 && (
   103→        <div className="flex items-center gap-1 mb-3 overflow-x-auto pb-1">
   104→          {data.sheetNames.map((sheetName) => (
   105→            <Button
   106→              key={sheetName}
   107→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   108→              size="sm"
   109→              className="text-xs whitespace-nowrap"
   110→              onClick={() => setActiveSheet(sheetName)}
   111→            >
   112→              {sheetName}
   113→            </Button>
   114→          ))}
   115→        </div>
   116→      )}
   117→
   118→      {/* Table */}
   119→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   120→        <table className="w-full text-sm">
   121→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   122→            <tr>
   123→              {sheetData.headers.map((header, i) => (
   124→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   125→                  {header || `Column ${i + 1}`}
   126→                </th>
   127→              ))}
   128→            </tr>
   129→          </thead>
   130→          <tbody>
   131→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   132→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   133→                {sheetData.headers.map((header, j) => (
   134→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   135→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   136→                  </td>
   137→                ))}
   138→              </tr>
   139→            ))}
   140→          </tbody>
   141→        </table>
   142→        {sheetData.rows.length > maxRows && (
   143→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   144→            Showing {maxRows} of {sheetData.rows.length} rows
   145→          </div>
   146→        )}
   147→        {sheetData.rows.length === 0 && (
   148→          <div className="p-6 text-center text-sm text-muted-foreground">
   149→            This sheet is empty
   150→          </div>
   151→        )}
   152→      </div>
   153→    </div>
   154→  );
   155→}

   222→// PDF Preview (Tier 1 — pdfjs-dist canvas rendering)
   223→// ============================================================
   224→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   225→  contentUrl: string;
   226→  name: string;
   227→  sizeBytes?: number;
   228→  mimeLabel: string;
   229→  closeButton: React.ReactNode;
   230→}) {
   231→  const [loading, setLoading] = useState(true);
   232→  const [error, setError] = useState(false);
   233→  const [numPages, setNumPages] = useState(0);
   234→  const [currentPage, setCurrentPage] = useState(1);
   235→  const [scale, setScale] = useState(1.2);
   236→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   237→  const containerRef = useRef<HTMLDivElement>(null);
   238→
   239→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   240→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   241→    const page = await pdfDocument.getPage(pageNum);
   242→    const viewport = page.getViewport({ scale });
   243→
   244→    let canvas = canvasRefs.current.get(pageNum);
   245→    if (!canvas) {
   246→      canvas = document.createElement('canvas');
   247→      canvasRefs.current.set(pageNum, canvas);
   248→      if (containerRef.current) {
   249→        containerRef.current.appendChild(canvas);
   250→      }
   251→    }
   252→
   253→    canvas.width = viewport.width;
   254→    canvas.height = viewport.height;
   255→    canvas.style.width = `${viewport.width}px`;
   256→    canvas.style.height = `${viewport.height}px`;
   257→    canvas.style.marginBottom = '8px';
   258→    canvas.style.display = 'block';
   259→
   260→    const context = canvas.getContext('2d');
   261→    if (!context) return;
   262→
   263→    await page.render({
   264→      canvasContext: context,
   265→      viewport,
   266→    });
   267→  }, []);
   268→
   269→  useEffect(() => {
   270→    let cancelled = false;
   271→
   272→    const loadPdf = async () => {
   273→      try {
   274→        // Dynamic import of pdfjs-dist
   275→        const pdfjsLib = await import('pdfjs-dist');
   276→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   277→
   278→        // Fetch PDF as ArrayBuffer from contentUrl
   279→        const res = await fetch(contentUrl);
   280→        if (!res.ok) throw new Error('Failed to fetch PDF');
   281→        const arrayBuffer = await res.arrayBuffer();
   282→
   283→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   284→        if (!cancelled) {
   285→          setNumPages(pdfDoc.numPages);
   286→          setLoading(false);
   287→        }
   288→
   289→        // Render all pages
   290→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   291→          if (cancelled) break;
   292→          await renderPage(i, pdfDoc, scale);
   293→        }
   294→      } catch (err) {
   295→        if (!cancelled) {
   296→          console.error('PDF preview error:', err);
   297→          setError(true);
   298→          setLoading(false);
   299→        }
   300→      }
   301→    };
   302→
   303→    loadPdf();
   304→    return () => { cancelled = true; };
   305→  }, [contentUrl, renderPage]);
   306→
   307→  // Re-render pages when scale changes
   308→  useEffect(() => {
   309→    if (numPages === 0 || loading) return;
   310→    let cancelled = false;
   311→
   312→    const rerender = async () => {
   313→      try {
   314→        const pdfjsLib = await import('pdfjs-dist');
   315→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   316→
   317→        const res = await fetch(contentUrl);
   318→        if (!res.ok) return;
   319→        const arrayBuffer = await res.arrayBuffer();
   320→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   321→
   322→        // Clear existing canvases
   323→        if (containerRef.current) {
   324→          containerRef.current.innerHTML = '';
   325→        }
   326→        canvasRefs.current.clear();
   327→
   328→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   329→          if (cancelled) break;
   330→          await renderPage(i, pdfDoc, scale);
   331→        }
   332→      } catch {
   333→        // Silently fail on re-render
   334→      }
   335→    };
   336→
   337→    rerender();
   338→    return () => { cancelled = true; };
   339→  }, [scale, numPages, contentUrl, renderPage]);
   340→
   341→  if (error) {
   342→    return (
   343→      <div className="relative flex flex-col w-full">
   344→        {closeButton}
   345→        <Card className="w-full">
   346→          <CardContent className="p-6 flex flex-col items-center gap-4">
   347→            <FileText className="h-12 w-12 text-muted-foreground" />
   348→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   349→            <Button variant="outline" size="sm" asChild>
   350→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   351→                <ExternalLink className="h-4 w-4 mr-2" />
   352→                Open PDF directly
   353→              </a>
   354→            </Button>
   355→          </CardContent>
   356→        </Card>
   357→      </div>
   358→    );
   359→  }
   360→
   361→  return (
   362→    <div className="relative flex flex-col w-full">
   363→      {closeButton}
   364→      <div className="flex items-center gap-2 mb-3">
   365→        <FileText className="h-5 w-5 text-red-500" />
   366→        <span className="font-medium truncate max-w-xs">{name}</span>
   367→        {sizeBytes && (
   368→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   369→        )}
   370→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   371→        <OfflineBadge />
   372→        <div className="flex items-center gap-1 ml-auto">
   373→          {numPages > 1 && (
   374→            <>
   375→              <Button
   376→                variant="outline"
   377→                size="icon"
   378→                className="h-7 w-7"
   379→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   380→                disabled={currentPage === 1}
   381→                aria-label="Previous page"
   382→              >
   383→                <ChevronLeft className="h-3 w-3" />
   384→              </Button>
   385→              <span className="text-xs text-muted-foreground">
   386→                Page {currentPage}/{numPages}
   387→              </span>
   388→              <Button
   389→                variant="outline"
   390→                size="icon"
   391→                className="h-7 w-7"
   392→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   393→                disabled={currentPage === numPages}
   394→                aria-label="Next page"
   395→              >
   396→                <ChevronRight className="h-3 w-3" />
   397→              </Button>
   398→            </>
   399→          )}
   400→          <Button
   401→            variant="outline"
   402→            size="icon"
   403→            className="h-7 w-7"
   404→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   405→            disabled={scale <= 0.5}
   406→            aria-label="Zoom out"
   407→          >
   408→            <ZoomOut className="h-3 w-3" />
   409→          </Button>
   410→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   411→          <Button
   412→            variant="outline"
   413→            size="icon"
   414→            className="h-7 w-7"
   415→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   416→            disabled={scale >= 3}
   417→            aria-label="Zoom in"
   418→          >
   419→            <ZoomIn className="h-3 w-3" />
   420→          </Button>
   421→        </div>
   422→      </div>
   423→
   424→      {loading ? (
   425→        <div className="flex items-center justify-center py-12">
   426→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   427→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   428→        </div>
   429→      ) : (
   430→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   431→          <div
   432→            ref={containerRef}
   433→            className="flex flex-col items-center p-4"
   434→          />
   435→        </ScrollArea>
   436→      )}
   437→    </div>
   438→  );
   439→}

   442→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   443→// MODUL 51: Added offline message support + cache integration
   444→// ============================================================
   445→function DocxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton, cachedContent, isFromCache, offlineMessage, triggerBackgroundCache, isLoadingCache }: {
   446→  contentUrl: string;
   447→  name: string;
   448→  sizeBytes?: number;
   449→  mimeLabel: string;
   450→  downloadUrl: string;
   451→  closeButton: React.ReactNode;
   452→  cachedContent: string | null;
   453→  isFromCache: boolean;
   454→  offlineMessage: string | null;
   455→  triggerBackgroundCache: (content: string) => Promise<void>;
   456→  isLoadingCache: boolean;
   457→}) {
   458→  const [loading, setLoading] = useState(true);
   459→  const [error, setError] = useState<string | null>(null);
   460→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   461→  const docxContainerRef = useRef<HTMLDivElement>(null);
   462→
   463→  // MODUL 51: If we have cached content and are offline or have a cache match, render it directly
   464→  const shouldUseCache = isFromCache && cachedContent;
   465→
   466→  useEffect(() => {
   467→    // MODUL 51: If we have cached HTML content, use it directly
   468→    if (shouldUseCache && cachedContent) {
   469→      setMammothHtml(cachedContent);
   470→      setLoading(false);
   471→      return;
   472→    }
   473→
   474→    let cancelled = false;
   475→
   476→    const loadDocx = async () => {
   477→      try {
   478→        // Fetch raw bytes from contentUrl
   479→        const res = await fetch(contentUrl);
   480→        if (!res.ok) throw new Error('Failed to fetch document');
   481→        const arrayBuffer = await res.arrayBuffer();
   482→
   483→        // Try docx-preview first
   484→        try {
   485→          const docxPreview = await import('docx-preview');
   486→          if (docxContainerRef.current && !cancelled) {
   487→            await docxPreview.renderAsync(
   488→              arrayBuffer,
   489→              docxContainerRef.current,
   490→              undefined,
   491→              { className: 'docx-preview-wrapper' }
   492→            );
   493→            if (!cancelled) {
   494→              setLoading(false);
   495→              // MODUL 51: Cache the rendered HTML — extract innerHTML from container for future offline use
   496→              const renderedHtml = docxContainerRef.current.innerHTML;
   497→              if (renderedHtml) {
   498→                triggerBackgroundCache(renderedHtml).catch((err) => {
   499→                  console.warn('[DocxPreview] Background cache write failed:', err);
   500→                });
   501→              }
   502→            }
   503→            return;
   504→          }
   505→        } catch (docxPreviewErr) {
   506→          console.warn('docx-preview failed, falling back to mammoth:', docxPreviewErr);
   507→        }
   508→
   509→        // Fallback to mammoth
   510→        if (!cancelled) {
   511→          const mammoth = await import('mammoth');
   512→          // In browser, mammoth expects {arrayBuffer: ArrayBuffer} not {buffer: Buffer}
   513→          const result = await mammoth.convertToHtml({ arrayBuffer: arrayBuffer });
   514→          if (!cancelled) {
   515→            setMammothHtml(result.value);
   516→            setLoading(false);
   517→            // MODUL 51: Cache the mammoth HTML result
   518→            if (result.value) {
   519→              triggerBackgroundCache(result.value).catch((err) => {
   520→                console.warn('[DocxPreview] Background cache write failed:', err);
   521→              });
   522→            }
   523→          }
   524→        }
   525→      } catch (err) {
   526→        if (!cancelled) {
   527→          console.error('DOCX preview error:', err);
   528→          setError('Failed to load document preview');
   529→          setLoading(false);
   530→        }
   531→      }
   532→    };
   533→
   534→    loadDocx();
   535→    return () => { cancelled = true; };
   536→  }, [contentUrl, shouldUseCache, cachedContent, triggerBackgroundCache]);
   537→
   538→  if (error) {
   539→    return (
   540→      <div className="relative flex flex-col w-full">
   541→        {closeButton}
   542→        <Card className="w-full">
   543→          <CardContent className="p-6 flex flex-col items-center gap-4">
   544→            <FileText className="h-12 w-12 text-muted-foreground" />
   545→            <p className="text-sm text-muted-foreground">{error}</p>
   546→            <Button variant="outline" size="sm" asChild>
   547→              <a href={downloadUrl} download={name}>
   548→                <Download className="h-4 w-4 mr-2" />
   549→                Download to open
   550→              </a>
   551→            </Button>
   552→          </CardContent>
   553→        </Card>
   554→      </div>
   555→    );
   556→  }
   557→
   558→  // MODUL 51: Offline with no cache — show explicit offline message
   559→  if (offlineMessage) {
   560→    return (
   561→      <div className="relative flex flex-col w-full">
   562→        {closeButton}
   563→        <Card className="w-full">
   564→          <CardContent className="p-6 flex flex-col items-center gap-4">
   565→            <WifiOff className="h-12 w-12 text-orange-500" />
   566→            <p className="text-sm text-orange-600 dark:text-orange-400 font-medium">{offlineMessage}</p>
   567→            <p className="text-xs text-muted-foreground text-center">
   568→              Open this file while online to make it available offline later
   569→            </p>
   570→          </CardContent>
   571→        </Card>
   572→      </div>
   573→    );
   574→  }
   575→
   576→  return (
   577→    <div className="relative flex flex-col w-full">
   578→      {closeButton}
   579→      <div className="flex items-center gap-2 mb-3">
   580→        <FileText className="h-5 w-5 text-blue-500" />
   581→        <span className="font-medium truncate max-w-xs">{name}</span>
   582→        {sizeBytes && (
   583→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   584→        )}
   585→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
