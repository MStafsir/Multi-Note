    81→// Spreadsheet Preview Sub-component
    82→// Renders xlsx data as a scrollable table with sheet tabs
    83→// ============================================================
    84→function SpreadsheetPreview({ data, name }: {
    85→  data: {
    86→    sheetNames: string[];
    87→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
    88→  };
    89→  name: string;
    90→}) {
    91→  const [activeSheet, setActiveSheet] = useState(data.sheetNames[0] || '');
    92→  const sheetData = data.sheets[activeSheet];
    93→  const maxRows = 100;
    94→
    95→  if (!sheetData) {
    96→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    97→  }
    98→
    99→  return (
   100→    <div className="flex flex-col w-full">
   101→      {/* Sheet tabs */}
   102→      {data.sheetNames.length > 1 && (
   103→        <div className="flex items-center gap-1 mb-3 overflow-x-auto pb-1">
   104→          {data.sheetNames.map((sheetName) => (
   105→            <Button
   106→              key={sheetName}
   107→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   108→              size="sm"
   109→              className="text-xs whitespace-nowrap"
   110→              onClick={() => setActiveSheet(sheetName)}
   111→            >
   112→              {sheetName}
   113→            </Button>
   114→          ))}
   115→        </div>
   116→      )}
   117→
   118→      {/* Table */}
   119→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   120→        <table className="w-full text-sm">
   121→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   122→            <tr>
   123→              {sheetData.headers.map((header, i) => (
   124→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   125→                  {header || `Column ${i + 1}`}
   126→                </th>
   127→              ))}
   128→            </tr>
   129→          </thead>
   130→          <tbody>
   131→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   132→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   133→                {sheetData.headers.map((header, j) => (
   134→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   135→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   136→                  </td>
   137→                ))}
   138→              </tr>
   139→            ))}
   140→          </tbody>
   141→        </table>
   142→        {sheetData.rows.length > maxRows && (
   143→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   144→            Showing {maxRows} of {sheetData.rows.length} rows
   145→          </div>
   146→        )}
   147→        {sheetData.rows.length === 0 && (
   148→          <div className="p-6 text-center text-sm text-muted-foreground">
   149→            This sheet is empty
   150→          </div>
   151→        )}
   152→      </div>
   153→    </div>
   154→  );
   155→}

   222→// PDF Preview (Tier 1 — pdfjs-dist canvas rendering)
   223→// ============================================================
   224→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   225→  contentUrl: string;
   226→  name: string;
   227→  sizeBytes?: number;
   228→  mimeLabel: string;
   229→  closeButton: React.ReactNode;
   230→}) {
   231→  const [loading, setLoading] = useState(true);
   232→  const [error, setError] = useState(false);
   233→  const [numPages, setNumPages] = useState(0);
   234→  const [currentPage, setCurrentPage] = useState(1);
   235→  const [scale, setScale] = useState(1.2);
   236→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   237→  const containerRef = useRef<HTMLDivElement>(null);
   238→
   239→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   240→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   241→    const page = await pdfDocument.getPage(pageNum);
   242→    const viewport = page.getViewport({ scale });
   243→
   244→    let canvas = canvasRefs.current.get(pageNum);
   245→    if (!canvas) {
   246→      canvas = document.createElement('canvas');
   247→      canvasRefs.current.set(pageNum, canvas);
   248→      if (containerRef.current) {
   249→        containerRef.current.appendChild(canvas);
   250→      }
   251→    }
   252→
   253→    canvas.width = viewport.width;
   254→    canvas.height = viewport.height;
   255→    canvas.style.width = `${viewport.width}px`;
   256→    canvas.style.height = `${viewport.height}px`;
   257→    canvas.style.marginBottom = '8px';
   258→    canvas.style.display = 'block';
   259→
   260→    const context = canvas.getContext('2d');
   261→    if (!context) return;
   262→
   263→    await page.render({
   264→      canvasContext: context,
   265→      viewport,
   266→    });
   267→  }, []);
   268→
   269→  useEffect(() => {
   270→    let cancelled = false;
   271→
   272→    const loadPdf = async () => {
   273→      try {
   274→        // Dynamic import of pdfjs-dist
   275→        const pdfjsLib = await import('pdfjs-dist');
   276→        // Use local worker from public dir (avoids CSP CDN blocking issues)
   277→        pdfjsLib.GlobalWorkerOptions.workerSrc = `/pdf.worker.min.mjs`;
   278→
   279→        // Fetch PDF as ArrayBuffer from contentUrl
   280→        const res = await fetch(contentUrl);
   281→        if (!res.ok) throw new Error('Failed to fetch PDF');
   282→        const arrayBuffer = await res.arrayBuffer();
   283→
   284→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   285→        if (!cancelled) {
   286→          setNumPages(pdfDoc.numPages);
   287→          setLoading(false);
   288→        }
   289→
   290→        // Render all pages
   291→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   292→          if (cancelled) break;
   293→          await renderPage(i, pdfDoc, scale);
   294→        }
   295→      } catch (err) {
   296→        if (!cancelled) {
   297→          console.error('PDF preview error:', err);
   298→          setError(true);
   299→          setLoading(false);
   300→        }
   301→      }
   302→    };
   303→
   304→    loadPdf();
   305→    return () => { cancelled = true; };
   306→  }, [contentUrl, renderPage]);
   307→
   308→  // Re-render pages when scale changes
   309→  useEffect(() => {
   310→    if (numPages === 0 || loading) return;
   311→    let cancelled = false;
   312→
   313→    const rerender = async () => {
   314→      try {
   315→        const pdfjsLib = await import('pdfjs-dist');
   316→        // Use local worker from public dir (avoids CSP CDN blocking issues)
   317→        pdfjsLib.GlobalWorkerOptions.workerSrc = `/pdf.worker.min.mjs`;
   318→
   319→        const res = await fetch(contentUrl);
   320→        if (!res.ok) return;
   321→        const arrayBuffer = await res.arrayBuffer();
   322→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   323→
   324→        // Clear existing canvases
   325→        if (containerRef.current) {
   326→          containerRef.current.innerHTML = '';
   327→        }
   328→        canvasRefs.current.clear();
   329→
   330→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   331→          if (cancelled) break;
   332→          await renderPage(i, pdfDoc, scale);
   333→        }
   334→      } catch {
   335→        // Silently fail on re-render
   336→      }
   337→    };
   338→
   339→    rerender();
   340→    return () => { cancelled = true; };
   341→  }, [scale, numPages, contentUrl, renderPage]);
   342→
   343→  if (error) {
   344→    return (
   345→      <div className="relative flex flex-col w-full">
   346→        {closeButton}
   347→        <Card className="w-full">
   348→          <CardContent className="p-6 flex flex-col items-center gap-4">
   349→            <FileText className="h-12 w-12 text-muted-foreground" />
   350→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   351→            <Button variant="outline" size="sm" asChild>
   352→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   353→                <ExternalLink className="h-4 w-4 mr-2" />
   354→                Open PDF directly
   355→              </a>
   356→            </Button>
   357→          </CardContent>
   358→        </Card>
   359→      </div>
   360→    );
   361→  }
   362→
   363→  return (
   364→    <div className="relative flex flex-col w-full">
   365→      {closeButton}
   366→      <div className="flex items-center gap-2 mb-3">
   367→        <FileText className="h-5 w-5 text-red-500" />
   368→        <span className="font-medium truncate max-w-xs">{name}</span>
   369→        {sizeBytes && (
   370→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   371→        )}
   372→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   373→        <OfflineBadge />
   374→        <div className="flex items-center gap-1 ml-auto">
   375→          {numPages > 1 && (
   376→            <>
   377→              <Button
   378→                variant="outline"
   379→                size="icon"
   380→                className="h-7 w-7"
   381→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   382→                disabled={currentPage === 1}
   383→                aria-label="Previous page"
   384→              >
   385→                <ChevronLeft className="h-3 w-3" />
   386→              </Button>
   387→              <span className="text-xs text-muted-foreground">
   388→                Page {currentPage}/{numPages}
   389→              </span>
   390→              <Button
   391→                variant="outline"
   392→                size="icon"
   393→                className="h-7 w-7"
   394→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   395→                disabled={currentPage === numPages}
   396→                aria-label="Next page"
   397→              >
   398→                <ChevronRight className="h-3 w-3" />
   399→              </Button>
   400→            </>
   401→          )}
   402→          <Button
   403→            variant="outline"
   404→            size="icon"
   405→            className="h-7 w-7"
   406→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   407→            disabled={scale <= 0.5}
   408→            aria-label="Zoom out"
   409→          >
   410→            <ZoomOut className="h-3 w-3" />
   411→          </Button>
   412→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   413→          <Button
   414→            variant="outline"
   415→            size="icon"
   416→            className="h-7 w-7"
   417→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   418→            disabled={scale >= 3}
   419→            aria-label="Zoom in"
   420→          >
   421→            <ZoomIn className="h-3 w-3" />
   422→          </Button>
   423→        </div>
   424→      </div>
   425→
   426→      {loading ? (
   427→        <div className="flex items-center justify-center py-12">
   428→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   429→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   430→        </div>
   431→      ) : (
   432→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   433→          <div
   434→            ref={containerRef}
   435→            className="flex flex-col items-center p-4"
   436→          />
   437→        </ScrollArea>
   438→      )}
   439→    </div>
   440→  );
   441→}

   444→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   445→// MODUL 51: Added offline message support + cache integration
   446→// ============================================================
   447→function DocxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton, cachedContent, isFromCache, offlineMessage, triggerBackgroundCache, isLoadingCache }: {
   448→  contentUrl: string;
   449→  name: string;
   450→  sizeBytes?: number;
   451→  mimeLabel: string;
   452→  downloadUrl: string;
   453→  closeButton: React.ReactNode;
   454→  cachedContent: string | null;
   455→  isFromCache: boolean;
   456→  offlineMessage: string | null;
   457→  triggerBackgroundCache: (content: string) => Promise<void>;
   458→  isLoadingCache: boolean;
   459→}) {
   460→  const [loading, setLoading] = useState(true);
   461→  const [error, setError] = useState<string | null>(null);
   462→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   463→  const docxContainerRef = useRef<HTMLDivElement>(null);
   464→
   465→  // MODUL 51: If we have cached content and are offline or have a cache match, render it directly
   466→  const shouldUseCache = isFromCache && cachedContent;
   467→
   468→  useEffect(() => {
   469→    // MODUL 51: If we have cached HTML content, use it directly
   470→    if (shouldUseCache && cachedContent) {
   471→      setMammothHtml(cachedContent);
   472→      setLoading(false);
   473→      return;
   474→    }
   475→
   476→    let cancelled = false;
   477→
   478→    const loadDocx = async () => {
   479→      try {
   480→        // Fetch raw bytes from contentUrl
   481→        const res = await fetch(contentUrl);
   482→        if (!res.ok) throw new Error('Failed to fetch document');
   483→        const arrayBuffer = await res.arrayBuffer();
   484→
   485→        // Try docx-preview first
   486→        try {
   487→          const docxPreview = await import('docx-preview');
   488→          if (docxContainerRef.current && !cancelled) {
   489→            await docxPreview.renderAsync(
   490→              arrayBuffer,
   491→              docxContainerRef.current,
   492→              undefined,
   493→              { className: 'docx-preview-wrapper' }
   494→            );
   495→            if (!cancelled) {
   496→              setLoading(false);
   497→              // MODUL 51: Cache the rendered HTML — extract innerHTML from container for future offline use
   498→              const renderedHtml = docxContainerRef.current.innerHTML;
   499→              if (renderedHtml) {
   500→                triggerBackgroundCache(renderedHtml).catch((err) => {
   501→                  console.warn('[DocxPreview] Background cache write failed:', err);
   502→                });
   503→              }
   504→            }
   505→            return;
   506→          }
   507→        } catch (docxPreviewErr) {
   508→          console.warn('docx-preview failed, falling back to mammoth:', docxPreviewErr);
   509→        }
   510→
   511→        // Fallback to mammoth
   512→        if (!cancelled) {
   513→          const mammoth = await import('mammoth');
   514→          // In browser, mammoth expects {arrayBuffer: ArrayBuffer} not {buffer: Buffer}
   515→          const result = await mammoth.convertToHtml({ arrayBuffer: arrayBuffer });
   516→          if (!cancelled) {
   517→            setMammothHtml(result.value);
   518→            setLoading(false);
   519→            // MODUL 51: Cache the mammoth HTML result
   520→            if (result.value) {
   521→              triggerBackgroundCache(result.value).catch((err) => {
   522→                console.warn('[DocxPreview] Background cache write failed:', err);
   523→              });
   524→            }
   525→          }
   526→        }
   527→      } catch (err) {
   528→        if (!cancelled) {
   529→          console.error('DOCX preview error:', err);
   530→          setError('Failed to load document preview');
   531→          setLoading(false);
   532→        }
   533→      }
   534→    };
   535→
   536→    loadDocx();
   537→    return () => { cancelled = true; };
   538→  }, [contentUrl, shouldUseCache, cachedContent, triggerBackgroundCache]);
   539→
   540→  if (error) {
   541→    return (
   542→      <div className="relative flex flex-col w-full">
   543→        {closeButton}
   544→        <Card className="w-full">
   545→          <CardContent className="p-6 flex flex-col items-center gap-4">
   546→            <FileText className="h-12 w-12 text-muted-foreground" />
   547→            <p className="text-sm text-muted-foreground">{error}</p>
   548→            <Button variant="outline" size="sm" asChild>
   549→              <a href={downloadUrl} download={name}>
   550→                <Download className="h-4 w-4 mr-2" />
   551→                Download to open
   552→              </a>
   553→            </Button>
   554→          </CardContent>
   555→        </Card>
   556→      </div>
   557→    );
   558→  }
   559→
   560→  // MODUL 51: Offline with no cache — show explicit offline message
   561→  if (offlineMessage) {
   562→    return (
   563→      <div className="relative flex flex-col w-full">
   564→        {closeButton}
   565→        <Card className="w-full">
   566→          <CardContent className="p-6 flex flex-col items-center gap-4">
   567→            <WifiOff className="h-12 w-12 text-orange-500" />
   568→            <p className="text-sm text-orange-600 dark:text-orange-400 font-medium">{offlineMessage}</p>
   569→            <p className="text-xs text-muted-foreground text-center">
   570→              Open this file while online to make it available offline later
   571→            </p>
   572→          </CardContent>
   573→        </Card>
   574→      </div>
   575→    );
   576→  }
   577→
   578→  return (
   579→    <div className="relative flex flex-col w-full">
   580→      {closeButton}
   581→      <div className="flex items-center gap-2 mb-3">
   582→        <FileText className="h-5 w-5 text-blue-500" />
   583→        <span className="font-medium truncate max-w-xs">{name}</span>
   584→        {sizeBytes && (
   585→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   586→        )}
   587→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
