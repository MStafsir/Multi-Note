   186→   186→// PDF Preview Sub-component (uses pdfjs-dist)
   187→   187→// Shared by both Tier 1 (native PDF) and High-Fidelity (LibreOffice→PDF)
   188→   188→// ============================================================
   189→   189→function PdfPreview({ contentUrl }: { contentUrl: string }) {
   190→   190→  const [numPages, setNumPages] = useState(0);
   191→   191→  const [pdfDoc, setPdfDoc] = useState<unknown>(null);
   192→   192→  const [scale, setScale] = useState(1.2);
   193→   193→  const [pdfError, setPdfError] = useState<string | null>(null);
   194→   194→  // MODUL 64.5: Explicit loading state for cache-miss scenario
   195→   195→  const [pdfLoading, setPdfLoading] = useState(true);
   196→   196→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   197→   197→
   198→   198→  useEffect(() => {
   199→   199→    let cancelled = false;
   200→   200→
   201→   201→    const loadPdf = async () => {
   202→   202→      try {
   203→   203→        setPdfLoading(true);
   204→   204→        const pdfjsLib = await import('pdfjs-dist');
   205→   205→        pdfjsLib.GlobalWorkerOptions.workerSrc = '/pdf.worker.min.mjs';
   206→   206→
   207→   207→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   208→   208→        if (!res.ok) throw new Error(`Failed to fetch PDF (${res.status})`);
   209→   209→        const arrayBuffer = await res.arrayBuffer();
   210→   210→        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   211→   211→
   212→   212→        if (!cancelled) {
   213→   213→          setPdfDoc(pdf);
   214→   214→          setNumPages(pdf.numPages);
   215→   215→          setPdfLoading(false);
   216→   216→        }
   217→   217→      } catch (err) {
   218→   218→        if (!cancelled) {
   219→   219→          console.error('PDF load error:', err);
   220→   220→          setPdfError('Failed to load PDF document');
   221→   221→          setPdfLoading(false);
   222→   222→        }
   223→   223→      }
   224→   224→    };
   225→   225→
   226→   226→    loadPdf();
   227→   227→    return () => { cancelled = true; };
   228→   228→  }, [contentUrl]);
   229→   229→
   230→   230→  // Render all pages when pdfDoc or scale changes
   231→   231→  useEffect(() => {
   232→   232→    if (!pdfDoc || numPages === 0) return;
   233→   233→    let cancelled = false;
   234→   234→
   235→   235→    const renderAllPages = async () => {
   236→   236→      const pdfjsLib = await import('pdfjs-dist');
   237→   237→
   238→   238→      for (let i = 1; i <= numPages; i++) {
   239→   239→        if (cancelled) break;
   240→   240→        try {
   241→   241→          const page = await (pdfDoc as { getPage: (n: number) => Promise<unknown> }).getPage(i);
   242→   242→          const viewport = (page as { getViewport: (opts: { scale: number }) => { width: number; height: number } }).getViewport({ scale });
   243→   243→          const canvas = canvasRefs.current.get(i);
   244→   244→          if (!canvas) continue;
   245→   245→
   246→   246→          canvas.height = viewport.height;
   247→   247→          canvas.width = viewport.width;
   248→   248→
   249→   249→          const context = canvas.getContext('2d');
   250→   250→          if (!context) continue;
   251→   251→
   252→   252→          const renderContext = { canvasContext: context, viewport };
   253→   253→          await (page as { render: (ctx: unknown) => Promise<unknown> }).render(renderContext).promise;
   254→   254→        } catch (err) {
   255→   255→          console.error(`PDF render error page ${i}:`, err);
   256→   256→        }
   257→   257→      }
   258→   258→    };
   259→   259→
   260→   260→    renderAllPages();
   261→   261→    return () => { cancelled = true; };
   262→   262→  }, [pdfDoc, numPages, scale]);
   263→   263→
   264→   264→  if (pdfError) {
   265→   265→    return (
   266→   266→      <div className="flex flex-col items-center gap-3 p-8">
   267→   267→        <AlertTriangle className="h-8 w-8 text-destructive" />
   268→   268→        <p className="text-sm text-muted-foreground">{pdfError}</p>
   269→   269→      </div>
   270→   270→    );
   271→   271→  }
   272→   272→
   273→   273→  // MODUL 64.5: Explicit loading state for cache-miss (first-time conversion)
   274→   274→  if (pdfLoading) {
   275→   275→    return (
   276→   276→      <div className="flex flex-col items-center justify-center py-20 gap-4">
   277→   277→        <Loader2 className="h-10 w-10 animate-spin text-muted-foreground" />
   278→   278→        <div className="text-center">
   279→   279→          <p className="text-sm font-medium text-foreground">Mengkonversi dokumen ke PDF…</p>
   280→   280→          <p className="text-xs text-muted-foreground mt-1">Proses ini memerlukan LibreOffice dan hanya terjadi sekali (cache-miss)</p>
   281→   281→        </div>
   282→   282→      </div>
   283→   283→    );
   284→   284→  }
   285→   285→
   286→   286→  return (
   287→   287→    <div className="flex flex-col items-center w-full">
   288→   288→      {/* MODUL 65: PDF Zoom Controls — dark-theme tokens */}
   289→   289→      <div className="flex items-center gap-2 mb-3 sticky top-0 z-10 bg-background/95 backdrop-blur py-2 px-4 rounded-lg shadow-sm border border-border">
   290→   290→        <span className="text-sm text-muted-foreground">
   291→   291→          {numPages} page{numPages !== 1 ? 's' : ''}
   292→   292→        </span>
   293→   293→        <span className="mx-2 text-border">|</span>
   294→   294→        <Button variant="outline" size="icon" className="h-8 w-8"
   295→   295→          onClick={() => setScale(Math.max(0.5, scale - 0.2))} disabled={scale <= 0.5}>
   296→   296→          <ZoomOut className="h-4 w-4" />
   297→   297→        </Button>
   298→   298→        <span className="text-sm text-muted-foreground">{Math.round(scale * 100)}%</span>
   299→   299→        <Button variant="outline" size="icon" className="h-8 w-8"
   300→   300→          onClick={() => setScale(Math.min(3, scale + 0.2))} disabled={scale >= 3}>
   301→   301→          <ZoomIn className="h-4 w-4" />
   302→   302→        </Button>
   303→   303→      </div>
   304→   304→
   305→   305→      {/* PDF Canvases — all pages rendered vertically */}
   306→   306→      <div className="flex flex-col items-center gap-4">
   307→   307→        {Array.from({ length: numPages }, (_, i) => i + 1).map((pageNum) => (
   308→   308→          <div key={pageNum} className="shadow-lg border bg-white">
   309→   309→            <canvas
   310→   310→              ref={(el) => {
   311→   311→                if (el) {
   312→   312→                  canvasRefs.current.set(pageNum, el);
   313→   313→                } else {
   314→   314→                  canvasRefs.current.delete(pageNum);
   315→   315→                }
   316→   316→              }}
   317→   317→              className="block"
   318→   318→            />
   319→   319→          </div>
   320→   320→        ))}
   321→   321→      </div>
   322→   322→    </div>
   323→   323→  );
   324→   324→}
