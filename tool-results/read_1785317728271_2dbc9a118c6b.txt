    84→    84→function SpreadsheetPreview({ data, name }: {
    85→    85→  data: {
    86→    86→    sheetNames: string[];
    87→    87→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;

    89→    89→  name: string;

    92→    92→  const sheetData = data.sheets[activeSheet];
    93→    93→  const maxRows = 100;
    94→    94→
    95→    95→  if (!sheetData) {
    96→    96→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    97→    97→  }
    98→    98→
    99→    99→  return (
   100→   100→    <div className="flex flex-col w-full">
   101→   101→      {/* Sheet tabs */}
   102→   102→      {data.sheetNames.length > 1 && (
   103→   103→        <div className="flex items-center gap-1 mb-3 overflow-x-auto pb-1">
   104→   104→          {data.sheetNames.map((sheetName) => (
   105→   105→            <Button
   106→   106→              key={sheetName}
   107→   107→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   108→   108→              size="sm"
   109→   109→              className="text-xs whitespace-nowrap"
   110→   110→              onClick={() => setActiveSheet(sheetName)}
   111→   111→            >
   112→   112→              {sheetName}
   113→   113→            </Button>
   114→   114→          ))}
   115→   115→        </div>
   116→   116→      )}
   117→   117→
   118→   118→      {/* Table */}
   119→   119→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   120→   120→        <table className="w-full text-sm">
   121→   121→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   122→   122→            <tr>
   123→   123→              {sheetData.headers.map((header, i) => (
   124→   124→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   125→   125→                  {header || `Column ${i + 1}`}
   126→   126→                </th>
   127→   127→              ))}
   128→   128→            </tr>
   129→   129→          </thead>
   130→   130→          <tbody>
   131→   131→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   132→   132→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   133→   133→                {sheetData.headers.map((header, j) => (
   134→   134→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   135→   135→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   136→   136→                  </td>
   137→   137→                ))}
   138→   138→              </tr>
   139→   139→            ))}
   140→   140→          </tbody>
   141→   141→        </table>
   142→   142→        {sheetData.rows.length > maxRows && (
   143→   143→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   144→   144→            Showing {maxRows} of {sheetData.rows.length} rows
   145→   145→          </div>
   146→   146→        )}
   147→   147→        {sheetData.rows.length === 0 && (
   148→   148→          <div className="p-6 text-center text-sm text-muted-foreground">
   149→   149→            This sheet is empty
   150→   150→          </div>
   151→   151→        )}
   152→   152→      </div>
   153→   153→    </div>
   154→   154→  );

   224→   224→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   225→   225→  contentUrl: string;
   226→   226→  name: string;
   227→   227→  sizeBytes?: number;
   228→   228→  mimeLabel: string;
   229→   229→  closeButton: React.ReactNode;
   230→   230→}) {
   231→   231→  const [loading, setLoading] = useState(true);
   232→   232→  const [error, setError] = useState(false);
   233→   233→  const [numPages, setNumPages] = useState(0);
   234→   234→  const [currentPage, setCurrentPage] = useState(1);
   235→   235→  const [scale, setScale] = useState(1.2);
   236→   236→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   237→   237→  const containerRef = useRef<HTMLDivElement>(null);
   238→   238→
   239→   239→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   240→   240→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   241→   241→    const page = await pdfDocument.getPage(pageNum);
   242→   242→    const viewport = page.getViewport({ scale });
   243→   243→
   244→   244→    let canvas = canvasRefs.current.get(pageNum);
   245→   245→    if (!canvas) {
   246→   246→      canvas = document.createElement('canvas');
   247→   247→      canvasRefs.current.set(pageNum, canvas);
   248→   248→      if (containerRef.current) {
   249→   249→        containerRef.current.appendChild(canvas);
   250→   250→      }
   251→   251→    }
   252→   252→
   253→   253→    canvas.width = viewport.width;
   254→   254→    canvas.height = viewport.height;
   255→   255→    canvas.style.width = `${viewport.width}px`;
   256→   256→    canvas.style.height = `${viewport.height}px`;
   257→   257→    canvas.style.marginBottom = '8px';
   258→   258→    canvas.style.display = 'block';
   259→   259→
   260→   260→    const context = canvas.getContext('2d');
   261→   261→    if (!context) return;
   262→   262→
   263→   263→    await page.render({
   264→   264→      canvasContext: context,
   265→   265→      viewport,
   266→   266→    });
   267→   267→  }, []);
   268→   268→
   269→   269→  useEffect(() => {
   270→   270→    let cancelled = false;
   271→   271→
   272→   272→    const loadPdf = async () => {
   273→   273→      try {
   274→   274→        // Dynamic import of pdfjs-dist
   275→   275→        const pdfjsLib = await import('pdfjs-dist');
   276→   276→        // Use local worker from public dir (avoids CSP CDN blocking issues)
   277→   277→        pdfjsLib.GlobalWorkerOptions.workerSrc = `/pdf.worker.min.mjs`;
   278→   278→
   279→   279→        // Fetch PDF as ArrayBuffer from contentUrl
   280→   280→        const res = await fetch(contentUrl);
   281→   281→        if (!res.ok) throw new Error('Failed to fetch PDF');
   282→   282→        const arrayBuffer = await res.arrayBuffer();
   283→   283→
   284→   284→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   285→   285→        if (!cancelled) {
   286→   286→          setNumPages(pdfDoc.numPages);
   287→   287→          setLoading(false);
   288→   288→        }
   289→   289→
   290→   290→        // Render all pages
   291→   291→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   292→   292→          if (cancelled) break;
   293→   293→          await renderPage(i, pdfDoc, scale);
   294→   294→        }
   295→   295→      } catch (err) {
   296→   296→        if (!cancelled) {
   297→   297→          console.error('PDF preview error:', err);
   298→   298→          setError(true);
   299→   299→          setLoading(false);
   300→   300→        }
   301→   301→      }
   302→   302→    };
   303→   303→
   304→   304→    loadPdf();
   305→   305→    return () => { cancelled = true; };
   306→   306→  }, [contentUrl, renderPage]);
   307→   307→
   308→   308→  // Re-render pages when scale changes
   309→   309→  useEffect(() => {
   310→   310→    if (numPages === 0 || loading) return;
   311→   311→    let cancelled = false;
   312→   312→
   313→   313→    const rerender = async () => {
   314→   314→      try {
   315→   315→        const pdfjsLib = await import('pdfjs-dist');
   316→   316→        // Use local worker from public dir (avoids CSP CDN blocking issues)
   317→   317→        pdfjsLib.GlobalWorkerOptions.workerSrc = `/pdf.worker.min.mjs`;
   318→   318→
   319→   319→        const res = await fetch(contentUrl);
   320→   320→        if (!res.ok) return;
   321→   321→        const arrayBuffer = await res.arrayBuffer();
   322→   322→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   323→   323→
   324→   324→        // Clear existing canvases
   325→   325→        if (containerRef.current) {
   326→   326→          containerRef.current.innerHTML = '';
   327→   327→        }
   328→   328→        canvasRefs.current.clear();
   329→   329→
   330→   330→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   331→   331→          if (cancelled) break;
   332→   332→          await renderPage(i, pdfDoc, scale);
   333→   333→        }
   334→   334→      } catch {
   335→   335→        // Silently fail on re-render
   336→   336→      }
   337→   337→    };
   338→   338→
   339→   339→    rerender();
   340→   340→    return () => { cancelled = true; };
   341→   341→  }, [scale, numPages, contentUrl, renderPage]);
   342→   342→
   343→   343→  if (error) {
   344→   344→    return (
   345→   345→      <div className="relative flex flex-col w-full">
   346→   346→        {closeButton}
   347→   347→        <Card className="w-full">
   348→   348→          <CardContent className="p-6 flex flex-col items-center gap-4">
   349→   349→            <FileText className="h-12 w-12 text-muted-foreground" />
   350→   350→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   351→   351→            <Button variant="outline" size="sm" asChild>
   352→   352→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   353→   353→                <ExternalLink className="h-4 w-4 mr-2" />
   354→   354→                Open PDF directly
   355→   355→              </a>
   356→   356→            </Button>
   357→   357→          </CardContent>
   358→   358→        </Card>
   359→   359→      </div>
   360→   360→    );
   361→   361→  }
   362→   362→
   363→   363→  return (
   364→   364→    <div className="relative flex flex-col w-full">
   365→   365→      {closeButton}
   366→   366→      <div className="flex items-center gap-2 mb-3">
   367→   367→        <FileText className="h-5 w-5 text-red-500" />
   368→   368→        <span className="font-medium truncate max-w-xs">{name}</span>
   369→   369→        {sizeBytes && (
   370→   370→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   371→   371→        )}
   372→   372→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   373→   373→        <OfflineBadge />
   374→   374→        <div className="flex items-center gap-1 ml-auto">
   375→   375→          {numPages > 1 && (
   376→   376→            <>
   377→   377→              <Button
   378→   378→                variant="outline"
   379→   379→                size="icon"
   380→   380→                className="h-7 w-7"
   381→   381→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   382→   382→                disabled={currentPage === 1}
   383→   383→                aria-label="Previous page"
   384→   384→              >
   385→   385→                <ChevronLeft className="h-3 w-3" />
   386→   386→              </Button>
   387→   387→              <span className="text-xs text-muted-foreground">
   388→   388→                Page {currentPage}/{numPages}
   389→   389→              </span>
   390→   390→              <Button
   391→   391→                variant="outline"
   392→   392→                size="icon"
   393→   393→                className="h-7 w-7"
   394→   394→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   395→   395→                disabled={currentPage === numPages}
   396→   396→                aria-label="Next page"
   397→   397→              >
   398→   398→                <ChevronRight className="h-3 w-3" />
   399→   399→              </Button>
   400→   400→            </>
   401→   401→          )}
   402→   402→          <Button
   403→   403→            variant="outline"
   404→   404→            size="icon"
   405→   405→            className="h-7 w-7"
   406→   406→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   407→   407→            disabled={scale <= 0.5}
   408→   408→            aria-label="Zoom out"
   409→   409→          >
   410→   410→            <ZoomOut className="h-3 w-3" />
   411→   411→          </Button>
   412→   412→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   413→   413→          <Button
   414→   414→            variant="outline"
   415→   415→            size="icon"
   416→   416→            className="h-7 w-7"
   417→   417→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   418→   418→            disabled={scale >= 3}
   419→   419→            aria-label="Zoom in"
   420→   420→          >
   421→   421→            <ZoomIn className="h-3 w-3" />
   422→   422→          </Button>
   423→   423→        </div>
   424→   424→      </div>
   425→   425→
   426→   426→      {loading ? (
   427→   427→        <div className="flex items-center justify-center py-12">
   428→   428→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   429→   429→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   430→   430→        </div>
   431→   431→      ) : (
   432→   432→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   433→   433→          <div
   434→   434→            ref={containerRef}
   435→   435→            className="flex flex-col items-center p-4"
   436→   436→          />
   437→   437→        </ScrollArea>
   438→   438→      )}
   439→   439→    </div>
   440→   440→  );
   441→   441→}

   444→   444→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   445→   445→// MODUL 51: Added offline message support + cache integration
   446→   446→// ============================================================
   447→   447→function DocxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton, cachedContent, isFromCache, offlineMessage, triggerBackgroundCache, isLoadingCache }: {
   448→   448→  contentUrl: string;
   449→   449→  name: string;
   450→   450→  sizeBytes?: number;
   451→   451→  mimeLabel: string;
   452→   452→  downloadUrl: string;
   453→   453→  closeButton: React.ReactNode;
   454→   454→  cachedContent: string | null;
   455→   455→  isFromCache: boolean;
   456→   456→  offlineMessage: string | null;
   457→   457→  triggerBackgroundCache: (content: string) => Promise<void>;
   458→   458→  isLoadingCache: boolean;
   459→   459→}) {
   460→   460→  const [loading, setLoading] = useState(true);
   461→   461→  const [error, setError] = useState<string | null>(null);
   462→   462→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   463→   463→  const docxContainerRef = useRef<HTMLDivElement>(null);
   464→   464→
   465→   465→  // MODUL 51: If we have cached content and are offline or have a cache match, render it directly
   466→   466→  const shouldUseCache = isFromCache && cachedContent;
   467→   467→
   468→   468→  useEffect(() => {
   469→   469→    // MODUL 51: If we have cached HTML content, use it directly
   470→   470→    if (shouldUseCache && cachedContent) {
   471→   471→      setMammothHtml(cachedContent);
   472→   472→      setLoading(false);
   473→   473→      return;
   474→   474→    }
   475→   475→
   476→   476→    let cancelled = false;
   477→   477→
   478→   478→    const loadDocx = async () => {
   479→   479→      try {
   480→   480→        // Fetch raw bytes from contentUrl
   481→   481→        const res = await fetch(contentUrl);
   482→   482→        if (!res.ok) throw new Error('Failed to fetch document');
   483→   483→        const arrayBuffer = await res.arrayBuffer();
   484→   484→
