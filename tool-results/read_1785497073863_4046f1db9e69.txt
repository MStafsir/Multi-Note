    96→function getInitialViewMode(): 'grid' | 'list' {
    97→  if (typeof window === 'undefined') return 'grid';
    98→  try {
    99→    const stored = localStorage.getItem(VIEW_MODE_KEY);
   100→    if (stored === 'grid' || stored === 'list') return stored;
   101→  } catch { /* localStorage not available */ }
   102→  return 'grid';
   103→}

   318→      const date = new Date(dateStr);
   319→      const now = new Date();
   320→      const isCurrentYear = date.getFullYear() === now.getFullYear();
