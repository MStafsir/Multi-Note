    82→// Spreadsheet Preview Sub-component
    83→// Renders xlsx data as a scrollable table with sheet tabs
    84→// ============================================================
    85→function SpreadsheetPreview({ data, name }: {
    86→  data: {
    87→    sheetNames: string[];
    88→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
    89→  };
    90→  name: string;
    91→}) {
    92→  const [activeSheet, setActiveSheet] = useState(data.sheetNames[0] || '');
    93→  const sheetData = data.sheets[activeSheet];
    94→  const maxRows = 200;
    95→
    96→  if (!sheetData) {
    97→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    98→  }
    99→
   100→  return (
   101→    <div className="flex flex-col w-full">
   102→      {/* Sheet tabs */}
   103→      {data.sheetNames.length > 1 && (
   104→        <div className="flex items-center gap-1 mb-3 overflow-x-auto pb-1">
   105→          {data.sheetNames.map((sheetName) => (
   106→            <Button
   107→              key={sheetName}
   108→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   109→              size="sm"
   110→              className="text-xs whitespace-nowrap"
   111→              onClick={() => setActiveSheet(sheetName)}
   112→            >
   113→              {sheetName}
   114→            </Button>
   115→          ))}
   116→        </div>
   117→      )}
   118→
   119→      {/* Table */}
   120→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   121→        <table className="w-full text-sm">
   122→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   123→            <tr>
   124→              {sheetData.headers.map((header, i) => (
   125→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   126→                  {header || `Column ${i + 1}`}
   127→                </th>
   128→              ))}
   129→            </tr>
   130→          </thead>
   131→          <tbody>
   132→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   133→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   134→                {sheetData.headers.map((header, j) => (
   135→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   136→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   137→                  </td>
   138→                ))}
   139→              </tr>
   140→            ))}
   141→          </tbody>
   142→        </table>
   143→        {sheetData.rows.length > maxRows && (
   144→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   145→            Showing {maxRows} of {sheetData.rows.length} rows
   146→          </div>
   147→        )}
   148→        {sheetData.rows.length === 0 && (
   149→          <div className="p-6 text-center text-sm text-muted-foreground">
   150→            This sheet is empty
   151→          </div>
   152→        )}
   153→      </div>
   154→    </div>
   155→  );
   156→}

   223→// PDF Preview (Tier 1 — pdfjs-dist canvas rendering)
   224→// ============================================================
   225→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   226→  contentUrl: string;
   227→  name: string;
   228→  sizeBytes?: number;
   229→  mimeLabel: string;
   230→  closeButton: React.ReactNode;
   231→}) {
   232→  const [loading, setLoading] = useState(true);
   233→  const [error, setError] = useState(false);
   234→  const [numPages, setNumPages] = useState(0);
   235→  const [currentPage, setCurrentPage] = useState(1);
   236→  const [scale, setScale] = useState(1.2);
   237→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   238→  const containerRef = useRef<HTMLDivElement>(null);
   239→
   240→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   241→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   242→    const page = await pdfDocument.getPage(pageNum);
   243→    const viewport = page.getViewport({ scale });
   244→
   245→    let canvas = canvasRefs.current.get(pageNum);
   246→    if (!canvas) {
   247→      canvas = document.createElement('canvas');
   248→      canvasRefs.current.set(pageNum, canvas);
   249→      if (containerRef.current) {
   250→        containerRef.current.appendChild(canvas);
   251→      }
   252→    }
   253→
   254→    canvas.width = viewport.width;
   255→    canvas.height = viewport.height;
   256→    canvas.style.width = `${viewport.width}px`;
   257→    canvas.style.height = `${viewport.height}px`;
   258→    canvas.style.marginBottom = '8px';
   259→    canvas.style.display = 'block';
   260→
   261→    const context = canvas.getContext('2d');
   262→    if (!context) return;
   263→
   264→    await page.render({
   265→      canvasContext: context,
   266→      viewport,
   267→    });
   268→  }, []);
   269→
   270→  useEffect(() => {
   271→    let cancelled = false;
   272→
   273→    const loadPdf = async () => {
   274→      try {
   275→        // Dynamic import of pdfjs-dist
   276→        const pdfjsLib = await import('pdfjs-dist');
   277→        // Use local worker from public dir (avoids CSP CDN blocking issues)
   278→        pdfjsLib.GlobalWorkerOptions.workerSrc = `/pdf.worker.min.mjs`;
   279→
   280→        // Fetch PDF as ArrayBuffer from contentUrl
   281→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   282→        if (!res.ok) throw new Error('Failed to fetch PDF');
   283→        const arrayBuffer = await res.arrayBuffer();
   284→
   285→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   286→        if (!cancelled) {
   287→          setNumPages(pdfDoc.numPages);
   288→          setLoading(false);
   289→        }
   290→
   291→        // Render all pages
   292→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   293→          if (cancelled) break;
   294→          await renderPage(i, pdfDoc, scale);
   295→        }
   296→      } catch (err) {
   297→        if (!cancelled) {
   298→          console.error('PDF preview error:', err);
   299→          setError(true);
   300→          setLoading(false);
   301→        }
   302→      }
   303→    };
   304→
   305→    loadPdf();
   306→    return () => { cancelled = true; };
   307→  }, [contentUrl, renderPage]);
   308→
   309→  // Re-render pages when scale changes
   310→  useEffect(() => {
   311→    if (numPages === 0 || loading) return;
   312→    let cancelled = false;
   313→
   314→    const rerender = async () => {
   315→      try {
   316→        const pdfjsLib = await import('pdfjs-dist');
   317→        // Use local worker from public dir (avoids CSP CDN blocking issues)
   318→        pdfjsLib.GlobalWorkerOptions.workerSrc = `/pdf.worker.min.mjs`;
   319→
   320→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   321→        if (!res.ok) return;
   322→        const arrayBuffer = await res.arrayBuffer();
   323→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   324→
   325→        // Clear existing canvases
   326→        if (containerRef.current) {
   327→          containerRef.current.innerHTML = '';
   328→        }
   329→        canvasRefs.current.clear();
   330→
   331→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   332→          if (cancelled) break;
   333→          await renderPage(i, pdfDoc, scale);
   334→        }
   335→      } catch {
   336→        // Silently fail on re-render
   337→      }
   338→    };
   339→
   340→    rerender();
   341→    return () => { cancelled = true; };
   342→  }, [scale, numPages, contentUrl, renderPage]);
   343→
   344→  if (error) {
   345→    return (
   346→      <div className="relative flex flex-col w-full">
   347→        {closeButton}
   348→        <Card className="w-full">
   349→          <CardContent className="p-6 flex flex-col items-center gap-4">
   350→            <FileText className="h-12 w-12 text-muted-foreground" />
   351→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   352→            <Button variant="outline" size="sm" asChild>
   353→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   354→                <ExternalLink className="h-4 w-4 mr-2" />
   355→                Open PDF directly
   356→              </a>
   357→            </Button>
   358→          </CardContent>
   359→        </Card>
   360→      </div>
   361→    );
   362→  }
   363→
   364→  return (
   365→    <div className="relative flex flex-col w-full">
   366→      {closeButton}
   367→      <div className="flex items-center gap-2 mb-3">
   368→        <FileText className="h-5 w-5 text-red-500" />
   369→        <span className="font-medium truncate max-w-xs">{name}</span>
   370→        {sizeBytes && (
   371→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   372→        )}
   373→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   374→        <OfflineBadge />
   375→        <div className="flex items-center gap-1 ml-auto">
   376→          {numPages > 1 && (
   377→            <>
   378→              <Button
   379→                variant="outline"
   380→                size="icon"
   381→                className="h-7 w-7"
   382→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   383→                disabled={currentPage === 1}
   384→                aria-label="Previous page"
   385→              >
   386→                <ChevronLeft className="h-3 w-3" />
   387→              </Button>
   388→              <span className="text-xs text-muted-foreground">
   389→                Page {currentPage}/{numPages}
   390→              </span>
   391→              <Button
   392→                variant="outline"
   393→                size="icon"
   394→                className="h-7 w-7"
   395→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   396→                disabled={currentPage === numPages}
   397→                aria-label="Next page"
   398→              >
   399→                <ChevronRight className="h-3 w-3" />
   400→              </Button>
   401→            </>
   402→          )}
   403→          <Button
   404→            variant="outline"
   405→            size="icon"
   406→            className="h-7 w-7"
   407→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   408→            disabled={scale <= 0.5}
   409→            aria-label="Zoom out"
   410→          >
   411→            <ZoomOut className="h-3 w-3" />
   412→          </Button>
   413→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   414→          <Button
   415→            variant="outline"
   416→            size="icon"
   417→            className="h-7 w-7"
   418→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   419→            disabled={scale >= 3}
   420→            aria-label="Zoom in"
   421→          >
   422→            <ZoomIn className="h-3 w-3" />
   423→          </Button>
   424→        </div>
   425→      </div>
   426→
   427→      {loading ? (
   428→        <div className="flex items-center justify-center py-12">
   429→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   430→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   431→        </div>
   432→      ) : (
   433→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   434→          <div
   435→            ref={containerRef}
   436→            className="flex flex-col items-center p-4"
   437→          />
   438→        </ScrollArea>
   439→      )}
   440→    </div>
   441→  );
   442→}

   445→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   446→// MODUL 51: Added offline message support + cache integration
   447→// ============================================================
   448→function DocxPreview({ id, contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton, cachedContent, isFromCache, offlineMessage, triggerBackgroundCache, isLoadingCache }: {
   449→  id: string;
   450→  contentUrl: string;
   451→  name: string;
   452→  sizeBytes?: number;
   453→  mimeLabel: string;
   454→  downloadUrl: string;
   455→  closeButton: React.ReactNode;
   456→  cachedContent: string | null;
   457→  isFromCache: boolean;
   458→  offlineMessage: string | null;
   459→  triggerBackgroundCache: (content: string) => Promise<void>;
   460→  isLoadingCache: boolean;
   461→}) {
   462→  const [loading, setLoading] = useState(true);
   463→  const [error, setError] = useState<string | null>(null);
   464→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   465→  const docxContainerRef = useRef<HTMLDivElement>(null);
   466→
   467→  // MODUL 51: If we have cached content and are offline or have a cache match, render it directly
   468→  const shouldUseCache = isFromCache && cachedContent;
   469→
   470→  useEffect(() => {
   471→    // MODUL 51: If we have cached HTML content, use it directly
   472→    if (shouldUseCache && cachedContent) {
   473→      setMammothHtml(cachedContent);
   474→      setLoading(false);
   475→      return;
   476→    }
   477→
   478→    let cancelled = false;
   479→
   480→    const loadDocx = async () => {
   481→      try {
   482→        // Fetch raw bytes from contentUrl
   483→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   484→        if (!res.ok) throw new Error('Failed to fetch document');
   485→        const arrayBuffer = await res.arrayBuffer();
   486→
   487→        // Try docx-preview first
   488→        try {
   489→          const docxPreview = await import('docx-preview');
   490→          if (docxContainerRef.current && !cancelled) {
   491→            await docxPreview.renderAsync(
   492→              arrayBuffer,
   493→              docxContainerRef.current,
   494→              undefined,
   495→              {
   496→                className: 'docx-preview-wrapper',
   497→                inWrapper: true,
   498→                hideWrapperOnPrint: false,
   499→                ignoreWidth: false,
   500→                ignoreHeight: false,
   501→                ignoreFonts: false,
   502→                breakPages: true,
   503→                ignoreLastRenderedPageBreak: true,
   504→                experimental: true, // enables tab stops calculation (Modul 62.1)
   505→                trimXmlDeclaration: true,
   506→                useBase64URL: true, // ensures images are embedded (Modul 62.1)
   507→                renderHeaders: true, // Modul 62.2
   508→                renderFooters: true, // Modul 62.2
   509→                renderFootnotes: true,
   510→                renderEndnotes: true,
   511→                renderAltChunks: true,
   512→                renderChanges: false,
   513→                renderComments: false,
   514→                debug: false,
   515→              }
   516→            );
   517→            if (!cancelled) {
   518→              setLoading(false);
   519→              // MODUL 51: Cache the rendered HTML — extract innerHTML from container for future offline use
   520→              const renderedHtml = docxContainerRef.current.innerHTML;
   521→              if (renderedHtml) {
   522→                triggerBackgroundCache(renderedHtml).catch((err) => {
   523→                  console.warn('[DocxPreview] Background cache write failed:', err);
   524→                });
   525→              }
   526→            }
   527→            return;
   528→          }
   529→        } catch (docxPreviewErr) {
   530→          console.warn('docx-preview failed, falling back to mammoth:', docxPreviewErr);
   531→        }
   532→
   533→        // Fallback to mammoth
   534→        if (!cancelled) {
   535→          const mammoth = await import('mammoth');
   536→          // In browser, mammoth expects {arrayBuffer: ArrayBuffer} not {buffer: Buffer}
   537→          const result = await mammoth.convertToHtml({ arrayBuffer: arrayBuffer });
   538→          if (!cancelled) {
   539→            setMammothHtml(result.value);
   540→            setLoading(false);
   541→            // MODUL 51: Cache the mammoth HTML result
   542→            if (result.value) {
   543→              triggerBackgroundCache(result.value).catch((err) => {
   544→                console.warn('[DocxPreview] Background cache write failed:', err);
   545→              });
   546→            }
   547→          }
   548→        }
   549→      } catch (err) {
   550→        if (!cancelled) {
   551→          console.error('DOCX preview error:', err);
   552→          setError('Failed to load document preview');
   553→          setLoading(false);
   554→        }
   555→      }
   556→    };
   557→
   558→    loadDocx();
   559→    return () => { cancelled = true; };
   560→  }, [contentUrl, shouldUseCache, cachedContent, triggerBackgroundCache]);
   561→
   562→  if (error) {
   563→    return (
   564→      <div className="relative flex flex-col w-full">
   565→        {closeButton}
   566→        <Card className="w-full">
   567→          <CardContent className="p-6 flex flex-col items-center gap-4">
   568→            <FileText className="h-12 w-12 text-muted-foreground" />
   569→            <p className="text-sm text-muted-foreground">{error}</p>
   570→            <Button variant="outline" size="sm" asChild>
   571→              <a href={downloadUrl} download={name}>
   572→                <Download className="h-4 w-4 mr-2" />
   573→                Download to open
   574→              </a>
   575→            </Button>
   576→          </CardContent>
   577→        </Card>
   578→      </div>
   579→    );
   580→  }
   581→
   582→  // MODUL 51: Offline with no cache — show explicit offline message
   583→  if (offlineMessage) {
   584→    return (
   585→      <div className="relative flex flex-col w-full">
   586→        {closeButton}
   587→        <Card className="w-full">
   588→          <CardContent className="p-6 flex flex-col items-center gap-4">

   750→            </Button>
   751→          </CardContent>
   752→        </Card>
   753→      </div>
   754→    );
   755→  }
   756→
   757→  // MODUL 51: Offline with no cache — show explicit offline message
   758→  if (offlineMessage) {
   759→    return (
   760→      <div className="relative flex flex-col w-full">
   761→        {closeButton}
   762→        <Card className="w-full">
   763→          <CardContent className="p-6 flex flex-col items-center gap-4">
   764→            <WifiOff className="h-12 w-12 text-orange-500" />
   765→            <p className="text-sm text-orange-600 dark:text-orange-400 font-medium">{offlineMessage}</p>
   766→            <p className="text-xs text-muted-foreground text-center">
