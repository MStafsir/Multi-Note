    85→const VIEW_MODE_KEY = 'app-view-mode';
    86→function getInitialViewMode(): 'grid' | 'list' {
    87→  if (typeof window === 'undefined') return 'grid';
    88→  try {
    89→    const stored = localStorage.getItem(VIEW_MODE_KEY);
    90→    if (stored === 'grid' || stored === 'list') return stored;
    91→  } catch { /* localStorage not available */ }
    92→  return 'grid';
    93→}
