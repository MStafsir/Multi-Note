   180→   180→// PDF Preview Sub-component (uses pdfjs-dist)
   181→   181→// Shared by both Tier 1 (native PDF) and High-Fidelity (LibreOffice→PDF)
   182→   182→// ============================================================
   183→   183→function PdfPreview({ contentUrl }: { contentUrl: string }) {
   184→   184→  const [numPages, setNumPages] = useState(0);
   185→   185→  const [pdfDoc, setPdfDoc] = useState<unknown>(null);
   186→   186→  const [scale, setScale] = useState(1.2);
   187→   187→  const [pdfError, setPdfError] = useState<string | null>(null);
   188→   188→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   189→   189→
   190→   190→  useEffect(() => {
   191→   191→    let cancelled = false;
   192→   192→
   193→   193→    const loadPdf = async () => {
   194→   194→      try {
   195→   195→        const pdfjsLib = await import('pdfjs-dist');
   196→   196→        pdfjsLib.GlobalWorkerOptions.workerSrc = '/pdf.worker.min.mjs';
   197→   197→
   198→   198→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   199→   199→        if (!res.ok) throw new Error(`Failed to fetch PDF (${res.status})`);
   200→   200→        const arrayBuffer = await res.arrayBuffer();
   201→   201→        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   202→   202→
   203→   203→        if (!cancelled) {
   204→   204→          setPdfDoc(pdf);
   205→   205→          setNumPages(pdf.numPages);
   206→   206→        }
   207→   207→      } catch (err) {
   208→   208→        if (!cancelled) {
   209→   209→          console.error('PDF load error:', err);
   210→   210→          setPdfError('Failed to load PDF document');
   211→   211→        }
   212→   212→      }
   213→   213→    };
   214→   214→
   215→   215→    loadPdf();
   216→   216→    return () => { cancelled = true; };
   217→   217→  }, [contentUrl]);
   218→   218→
   219→   219→  // Render all pages when pdfDoc or scale changes
   220→   220→  useEffect(() => {
   221→   221→    if (!pdfDoc || numPages === 0) return;
   222→   222→    let cancelled = false;
   223→   223→
   224→   224→    const renderAllPages = async () => {
   225→   225→      const pdfjsLib = await import('pdfjs-dist');
   226→   226→
   227→   227→      for (let i = 1; i <= numPages; i++) {
   228→   228→        if (cancelled) break;
   229→   229→        try {
   230→   230→          const page = await (pdfDoc as { getPage: (n: number) => Promise<unknown> }).getPage(i);
   231→   231→          const viewport = (page as { getViewport: (opts: { scale: number }) => { width: number; height: number } }).getViewport({ scale });
   232→   232→          const canvas = canvasRefs.current.get(i);
   233→   233→          if (!canvas) continue;
   234→   234→
   235→   235→          canvas.height = viewport.height;
   236→   236→          canvas.width = viewport.width;
   237→   237→
   238→   238→          const context = canvas.getContext('2d');
   239→   239→          if (!context) continue;
   240→   240→
   241→   241→          const renderContext = { canvasContext: context, viewport };
   242→   242→          await (page as { render: (ctx: unknown) => Promise<unknown> }).render(renderContext).promise;
   243→   243→        } catch (err) {
   244→   244→          console.error(`PDF render error page ${i}:`, err);
   245→   245→        }
   246→   246→      }
   247→   247→    };
   248→   248→
   249→   249→    renderAllPages();
   250→   250→    return () => { cancelled = true; };
   251→   251→  }, [pdfDoc, numPages, scale]);
   252→   252→
   253→   253→  if (pdfError) {
   254→   254→    return (
   255→   255→      <div className="flex flex-col items-center gap-3 p-8">
   256→   256→        <AlertTriangle className="h-8 w-8 text-destructive" />
   257→   257→        <p className="text-sm text-muted-foreground">{pdfError}</p>
   258→   258→      </div>
   259→   259→    );
   260→   260→  }
   261→   261→
   262→   262→  return (
   263→   263→    <div className="flex flex-col items-center w-full">
   264→   264→      {/* PDF Zoom Controls */}
   265→   265→      <div className="flex items-center gap-2 mb-3 sticky top-0 z-10 bg-white/95 backdrop-blur py-2 px-4 rounded-lg shadow-sm">
   266→   266→        <span className="text-sm text-gray-600">
   267→   267→          {numPages} page{numPages !== 1 ? 's' : ''}
   268→   268→        </span>
   269→   269→        <span className="mx-2 text-gray-300">|</span>
   270→   270→        <Button variant="outline" size="icon" className="h-8 w-8"
   271→   271→          onClick={() => setScale(Math.max(0.5, scale - 0.2))} disabled={scale <= 0.5}>
   272→   272→          <ZoomOut className="h-4 w-4" />
   273→   273→        </Button>
   274→   274→        <span className="text-sm text-gray-600">{Math.round(scale * 100)}%</span>
   275→   275→        <Button variant="outline" size="icon" className="h-8 w-8"
   276→   276→          onClick={() => setScale(Math.min(3, scale + 0.2))} disabled={scale >= 3}>
   277→   277→          <ZoomIn className="h-4 w-4" />
   278→   278→        </Button>
   279→   279→      </div>
   280→   280→
   281→   281→      {/* PDF Canvases — all pages rendered vertically */}
   282→   282→      <div className="flex flex-col items-center gap-4">
   283→   283→        {Array.from({ length: numPages }, (_, i) => i + 1).map((pageNum) => (
   284→   284→          <div key={pageNum} className="shadow-lg border bg-white">
   285→   285→            <canvas
   286→   286→              ref={(el) => {
   287→   287→                if (el) {
   288→   288→                  canvasRefs.current.set(pageNum, el);
   289→   289→                } else {
   290→   290→                  canvasRefs.current.delete(pageNum);
   291→   291→                }
   292→   292→              }}
   293→   293→              className="block"
   294→   294→            />
   295→   295→          </div>
   296→   296→        ))}
   297→   297→      </div>
   298→   298→    </div>
   299→   299→  );
   300→   300→}

   449→   449→            );
   450→   450→            if (!cancelled) {

   452→   452→              setLoading(false);

   455→   455→          }

   482→   482→        }
   483→   483→      }

   485→   485→    loadDocx();
   486→   486→    return () => { cancelled = true; };
   487→   487→  }, [contentUrl, previewType]);
   488→   488→
   489→   489→  // ---- XLSX loading (Tier 2 — SheetJS client-side) ----

   509→   509→          const jsonData = XLSX.utils.sheet_to_json<Record<string, string | number | boolean | null>>(worksheet, { defval: null, raw: false });
   510→   510→          const headers = jsonData.length > 0 ? Object.keys(jsonData[0]) : [];
   511→   511→          sheets[sheetName] = { rows: jsonData.slice(0, 200), headers };
   512→   512→        }
   513→   513→
   514→   514→        if (!cancelled) {
   515→   515→          setSpreadsheetData({ sheetNames, sheets });
   516→   516→          setLoading(false);
   517→   517→        }

   521→   521→          setError('Failed to load spreadsheet preview. The file may be corrupted or password-protected.');
   522→   522→          setLoading(false);
   523→   523→        }
   524→   524→      }

   527→   527→    return () => { cancelled = true; };
   528→   528→  }, [contentUrl, previewType]);
   529→   529→
   530→   530→  // ---- PPTX loading (Tier 3 — server-side JSON) ----

   546→   546→          setLoading(false);
   547→   547→        }
   548→   548→      } catch (err) {

   550→   550→        if (!cancelled) {
   551→   551→          setError('Failed to load presentation preview');
   552→   552→          setLoading(false);
   553→   553→        }
   554→   554→      }
   555→   555→    };
   556→   556→    loadPptx();
   557→   557→    return () => { cancelled = true; };

   565→   565→    // The PdfPreview component handles its own loading state

   568→   568→

   621→   621→                <Button variant="outline" size="sm" asChild>

   631→   631→
   632→   632→  // ---- Main rendering ----
   633→   633→  return (

   637→   637→        <div className="flex items-center gap-2">
   638→   638→          <Button variant="ghost" size="icon" onClick={handleClose}><ArrowLeft className="h-5 w-5" /></Button>

   640→   640→          <span className="font-medium truncate max-w-[50vw]">{name}</span>

   733→   733→              {textContent}

   735→   735→          </div>

   740→   740→              <div
