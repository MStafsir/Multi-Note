   135→    </div>

   186→// PDF Preview Sub-component (uses pdfjs-dist)
   187→// Shared by both Tier 1 (native PDF) and High-Fidelity (LibreOffice→PDF)
   188→// ============================================================
   189→function PdfPreview({ contentUrl }: { contentUrl: string }) {
   190→  const [numPages, setNumPages] = useState(0);
   191→  const [pdfDoc, setPdfDoc] = useState<unknown>(null);
   192→  const [scale, setScale] = useState(1.2);
   193→  const [pdfError, setPdfError] = useState<string | null>(null);
   194→  // MODUL 64.5: Explicit loading state for cache-miss scenario
   195→  const [pdfLoading, setPdfLoading] = useState(true);
   196→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   197→
   198→  useEffect(() => {
   199→    let cancelled = false;
   200→
   201→    const loadPdf = async () => {
   202→      try {
   203→        setPdfLoading(true);
   204→        const pdfjsLib = await import('pdfjs-dist');
   205→        pdfjsLib.GlobalWorkerOptions.workerSrc = '/pdf.worker.min.mjs';
   206→
   207→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   208→        if (!res.ok) throw new Error(`Failed to fetch PDF (${res.status})`);
   209→        const arrayBuffer = await res.arrayBuffer();
   210→        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   211→
   212→        if (!cancelled) {
   213→          setPdfDoc(pdf);
   214→          setNumPages(pdf.numPages);
   215→          setPdfLoading(false);
   216→        }
   217→      } catch (err) {
   218→        if (!cancelled) {
   219→          console.error('PDF load error:', err);
   220→          setPdfError('Failed to load PDF document');
   221→          setPdfLoading(false);
   222→        }
   223→      }
   224→    };
   225→
   226→    loadPdf();
   227→    return () => { cancelled = true; };
   228→  }, [contentUrl]);
   229→
   230→  // Render all pages when pdfDoc or scale changes
   231→  useEffect(() => {
   232→    if (!pdfDoc || numPages === 0) return;
   233→    let cancelled = false;
   234→
   235→    const renderAllPages = async () => {
   236→      const pdfjsLib = await import('pdfjs-dist');
   237→
   238→      for (let i = 1; i <= numPages; i++) {
   239→        if (cancelled) break;
   240→        try {
   241→          const page = await (pdfDoc as { getPage: (n: number) => Promise<unknown> }).getPage(i);
   242→          const viewport = (page as { getViewport: (opts: { scale: number }) => { width: number; height: number } }).getViewport({ scale });
   243→          const canvas = canvasRefs.current.get(i);
   244→          if (!canvas) continue;
   245→
   246→          canvas.height = viewport.height;
   247→          canvas.width = viewport.width;
   248→
   249→          const context = canvas.getContext('2d');
   250→          if (!context) continue;
   251→
   252→          const renderContext = { canvasContext: context, viewport };
   253→          await (page as { render: (ctx: unknown) => Promise<unknown> }).render(renderContext).promise;
   254→        } catch (err) {
   255→          console.error(`PDF render error page ${i}:`, err);
   256→        }
   257→      }
   258→    };
   259→
   260→    renderAllPages();
   261→    return () => { cancelled = true; };
   262→  }, [pdfDoc, numPages, scale]);
   263→
   264→  if (pdfError) {
   265→    return (
   266→      <div className="flex flex-col items-center gap-3 p-8">
   267→        <AlertTriangle className="h-8 w-8 text-destructive" />
   268→        <p className="text-sm text-muted-foreground">{pdfError}</p>
   269→      </div>
   270→    );
   271→  }
   272→
   273→  // MODUL 64.5: Explicit loading state for cache-miss (first-time conversion)
   274→  if (pdfLoading) {
   275→    return (
   276→      <div className="flex flex-col items-center justify-center py-20 gap-4">
   277→        <Loader2 className="h-10 w-10 animate-spin text-muted-foreground" />
   278→        <div className="text-center">
   279→          <p className="text-sm font-medium text-foreground">Mengkonversi dokumen ke PDF…</p>
   280→          <p className="text-xs text-muted-foreground mt-1">Proses ini memerlukan LibreOffice dan hanya terjadi sekali (cache-miss)</p>
   281→        </div>
   282→      </div>
   283→    );
   284→  }
   285→
   286→  return (
   287→    <div className="flex flex-col items-center w-full">
   288→      {/* MODUL 65: PDF Zoom Controls — dark-theme tokens */}
   289→      <div className="flex items-center gap-2 mb-3 sticky top-0 z-10 bg-background/95 backdrop-blur py-2 px-4 rounded-lg shadow-sm border border-border">
   290→        <span className="text-sm text-muted-foreground">
   291→          {numPages} page{numPages !== 1 ? 's' : ''}
   292→        </span>
   293→        <span className="mx-2 text-border">|</span>
   294→        <Button variant="outline" size="icon" className="h-8 w-8"
   295→          onClick={() => setScale(Math.max(0.5, scale - 0.2))} disabled={scale <= 0.5}>
   296→          <ZoomOut className="h-4 w-4" />
   297→        </Button>
   298→        <span className="text-sm text-muted-foreground">{Math.round(scale * 100)}%</span>
   299→        <Button variant="outline" size="icon" className="h-8 w-8"
   300→          onClick={() => setScale(Math.min(3, scale + 0.2))} disabled={scale >= 3}>
   301→          <ZoomIn className="h-4 w-4" />
   302→        </Button>
   303→      </div>
   304→
   305→      {/* PDF Canvases — all pages rendered vertically */}
   306→      <div className="flex flex-col items-center gap-4">
   307→        {Array.from({ length: numPages }, (_, i) => i + 1).map((pageNum) => (
   308→          <div key={pageNum} className="shadow-lg border bg-white">
   309→            <canvas
   310→              ref={(el) => {
   311→                if (el) {
   312→                  canvasRefs.current.set(pageNum, el);
   313→                } else {
   314→                  canvasRefs.current.delete(pageNum);
   315→                }
   316→              }}
   317→              className="block"
   318→            />
   319→          </div>
   320→        ))}
   321→      </div>
   322→    </div>
   323→  );
   324→}

   326→// ============================================================
   327→// Main DedicatedViewer Component
   328→// MODUL 63: Two rendering paths — Default (fast) and High-Fidelity (PDF)
   329→// ============================================================
   330→export function DedicatedViewer({
   331→  nodeId,
   332→  name,
   333→  mimeType,
   334→  previewType,
   335→  previewTier,
   336→  mimeLabel,
   337→  sizeBytes,
   338→  checksumSha256,
   339→}: DedicatedViewerProps) {
   340→  const router = useRouter();
   341→  const [loading, setLoading] = useState(true);
   342→  const [error, setError] = useState<string | null>(null);
   343→
   344→  // MODUL 64.2: Default to high-fidelity (LibreOffice → PDF) for office types
   345→  // This is the confirmed accurate path — user no longer needs to click manually
   346→  const [hiFiMode, setHiFiMode] = useState<boolean>(() => HI_FI_TYPES.includes(previewType));
   347→  const [hiFiError, setHiFiError] = useState<string | null>(null);
   348→
   349→  // Whether this file type supports high-fidelity rendering
   350→  const supportsHiFi = HI_FI_TYPES.includes(previewType);
   351→
   352→  // DOCX state
   353→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   354→  const [docxRenderMethod, setDocxRenderMethod] = useState<'docx-preview' | 'mammoth' | null>(null);
   355→  const docxContainerRef = useRef<HTMLDivElement>(null);
   356→
   357→  // XLSX state
   358→  const [spreadsheetData, setSpreadsheetData] = useState<{
   359→    sheetNames: string[];
   360→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
   361→  } | null>(null);
   362→
   363→  // PPTX state
   364→  const [presentationData, setPresentationData] = useState<{
   365→    slideTexts: string[];
   366→    totalSlides: number;
   367→  } | null>(null);
   368→
   369→  // Text/Code state
   370→  const [textContent, setTextContent] = useState<string | null>(null);
   371→
   372→  // URLs — all same-origin, session-authenticated
   373→  const contentUrl = `/api/files/${nodeId}/content`;
   374→  const previewUrl = `/api/preview/${nodeId}`;
   375→  const downloadUrl = `/api/files/${nodeId}/content?download=true`;
   376→  const convertPdfUrl = `/api/files/${nodeId}/convert-pdf`;
   377→
   378→  // ---- High-Fidelity toggle handler ----
   379→  const handleHiFiToggle = () => {
   380→    if (hiFiMode) {
   381→      // Switch to fast mode (secondary)
   382→      setHiFiMode(false);
   383→      setHiFiError(null);
   384→    } else {
   385→      // Switch to high-fidelity mode (default)
   386→      setHiFiMode(true);
   387→      setHiFiError(null);
   388→    }
   389→  };
   390→
   391→  // ---- Tier 1: Text/Code loading ----
   392→  useEffect(() => {
   393→    if (previewType !== 'text' && previewType !== 'code') return;
   394→
   395→    let cancelled = false;
   396→    const loadText = async () => {
   397→      try {
   398→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   399→        if (!res.ok) throw new Error(`Failed to fetch file content (${res.status})`);
   400→        const text = await res.text();
   401→        if (!cancelled) {
   402→          setTextContent(text);
   403→          setLoading(false);
   404→        }
   405→      } catch (err) {
   406→        if (!cancelled) {
   407→          setError('Failed to load text content');
   408→          setLoading(false);
   409→        }
   410→      }
   411→    };
   412→    loadText();
   413→    return () => { cancelled = true; };
   414→  }, [contentUrl, previewType]);
   415→
   416→  // ---- Tier 1: Image/Video/Audio — loading is instant ----
   417→  useEffect(() => {
   418→    if (previewType === 'image' || previewType === 'video' || previewType === 'audio') {
   419→      setLoading(false);
   420→    }
   421→  }, [previewType]);
   422→
   423→  // ---- Tier 1: PDF — handled by PdfPreview sub-component ----
   424→  useEffect(() => {
   425→    if (previewType === 'pdf') {
   426→      setLoading(false); // PdfPreview handles its own loading
   427→    }
   428→  }, [previewType]);
   429→
   430→  // ---- DOCX loading (Tier 2 — docx-preview PRIMARY, mammoth fallback) ----
   431→  // MODUL 62: Full docx-preview config flags
   432→  useEffect(() => {
   433→    if (previewType !== 'docx') return;
   434→    let cancelled = false;
   435→
   436→    const loadDocx = async () => {
   437→      try {
   438→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   439→        if (!res.ok) throw new Error(`Failed to fetch document (${res.status})`);
   440→        const arrayBuffer = await res.arrayBuffer();
   441→
   442→        // PRIMARY: docx-preview — preserves images, headings, tables, lists, formatting
   443→        // MODUL 62: Full config flags for maximum fidelity
   444→        try {
   445→          const docxPreview = await import('docx-preview');
   446→          if (docxContainerRef.current && !cancelled) {
   447→            await docxPreview.renderAsync(
   448→              arrayBuffer,
   449→              docxContainerRef.current,
   450→              undefined,
   451→              {
   452→                className: 'docx-preview-wrapper',
   453→                inWrapper: true,
   454→                hideWrapperOnPrint: false,
   455→                ignoreWidth: false,
   456→                ignoreHeight: false,
   457→                ignoreFonts: false,
   458→                breakPages: true,
   459→                ignoreLastRenderedPageBreak: true,
   460→                experimental: true, // enables tab stops calculation (Modul 62.1)
   461→                trimXmlDeclaration: true,
   462→                useBase64URL: true, // ensures images are embedded (Modul 62.1)
   463→                renderHeaders: true, // Modul 62.2
   464→                renderFooters: true, // Modul 62.2
   465→                renderFootnotes: true,
   466→                renderEndnotes: true,
   467→                renderAltChunks: true,
   468→                renderChanges: false,
   469→                renderComments: false,
   470→                debug: false,
   471→              }
   472→            );
   473→            if (!cancelled) {
   474→              setDocxRenderMethod('docx-preview');
   475→              setLoading(false);
   476→            }
   477→            return;
   478→          }
   479→        } catch (docxPreviewErr) {
   480→          console.warn('[DedicatedViewer] docx-preview failed, falling back to mammoth:', docxPreviewErr);
   481→        }
   482→
   483→        // FALLBACK: mammoth — semantic HTML conversion (may lose images/formatting)
   484→        if (!cancelled) {
   485→          try {
   486→            const mammoth = await import('mammoth');
   487→            const result = await mammoth.convertToHtml({ arrayBuffer });
   488→            if (!cancelled) {
   489→              setMammothHtml(result.value);
   490→              setDocxRenderMethod('mammoth');
   491→              setLoading(false);
   492→            }
   493→          } catch (mammothErr) {
   494→            console.error('[DedicatedViewer] mammoth fallback also failed:', mammothErr);
   495→            if (!cancelled) {
   496→              setError('Failed to render document preview. Both docx-preview and mammoth failed.');
   497→              setLoading(false);
   498→            }
   499→          }
   500→        }
   501→      } catch (err) {
   502→        if (!cancelled) {
   503→          setError('Failed to load document file');
   504→          setLoading(false);
   505→        }
   506→      }
   507→    };
   508→    loadDocx();
   509→    return () => { cancelled = true; };
   510→  }, [contentUrl, previewType]);
   511→
   512→  // ---- XLSX loading (Tier 2 — SheetJS client-side) ----
   513→  useEffect(() => {
   514→    if (previewType !== 'xlsx') return;
   515→    let cancelled = false;
   516→
   517→    const loadXlsx = async () => {
   518→      try {
   519→        const res = await fetch(contentUrl, { credentials: 'same-origin' });
   520→        if (!res.ok) throw new Error(`Failed to fetch spreadsheet (${res.status})`);
   521→        const arrayBuffer = await res.arrayBuffer();
   522→
   523→        const XLSX = await import('xlsx');
   524→        const workbook = XLSX.read(arrayBuffer, { type: 'array' });
   525→        const sheetNames = workbook.SheetNames;
   526→        const sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }> = {};
   527→
   528→        for (const sheetName of sheetNames) {
   529→          const worksheet = workbook.Sheets[sheetName];
   530→          if (!worksheet) continue;
   531→          // raw: false → computed values (not formula strings)
   532→          const jsonData = XLSX.utils.sheet_to_json<Record<string, string | number | boolean | null>>(worksheet, { defval: null, raw: false });
   533→          const headers = jsonData.length > 0 ? Object.keys(jsonData[0]) : [];
   534→          sheets[sheetName] = { rows: jsonData.slice(0, 200), headers };
   535→        }
   536→
   537→        if (!cancelled) {
   538→          setSpreadsheetData({ sheetNames, sheets });
   539→          setLoading(false);
   540→        }
   541→      } catch (err) {
   542→        console.error('[DedicatedViewer] XLSX load error:', err);
   543→        if (!cancelled) {
   544→          setError('Failed to load spreadsheet preview. The file may be corrupted or password-protected.');
   545→          setLoading(false);
   546→        }
   547→      }
   548→    };
   549→    loadXlsx();
   550→    return () => { cancelled = true; };
   551→  }, [contentUrl, previewType]);
   552→
   553→  // ---- PPTX loading (Tier 3 — server-side JSON) ----
   554→  useEffect(() => {
   555→    if (previewType !== 'pptx') return;
   556→    let cancelled = false;
   557→
   558→    const loadPptx = async () => {
   559→      try {
   560→        const res = await fetch(previewUrl, { credentials: 'same-origin' });
   561→        if (!res.ok) throw new Error(`Failed to load presentation preview (${res.status})`);
   562→        const data = await res.json();
   563→        if (!cancelled) {
   564→          if (data.success && data.data) {
   565→            setPresentationData({ slideTexts: data.data.slideTexts || [], totalSlides: data.data.totalSlides || 0 });
   566→          } else {
   567→            setError('No preview data available for this presentation');
   568→          }
   569→          setLoading(false);
   570→        }
   571→      } catch (err) {
   572→        console.error('[DedicatedViewer] PPTX load error:', err);
   573→        if (!cancelled) {
   574→          setError('Failed to load presentation preview');
   575→          setLoading(false);
   576→        }
   577→      }
   578→    };
   579→    loadPptx();
   580→    return () => { cancelled = true; };
   581→  }, [previewUrl, previewType]);
   582→
   583→  // ---- MODUL 64.5: High-Fidelity mode loading ----
   584→  // The PdfPreview sub-component handles its own loading state (pdfLoading)
   585→  // with explicit progress indicators for cache-miss scenarios.
   586→  // No additional loading state needed here — PdfPreview shows its own
   587→  // "Mengkonversi dokumen ke PDF…" indicator during first-time conversion.
   588→
   589→  // ---- Close/back handler ----
   590→  const handleClose = () => {
   591→    if (window.opener) { window.close(); } else { router.back(); }
   592→  };
   593→
   594→  // ---- Icon selection ----
   595→  const getIcon = () => {
   596→    switch (previewType) {
   597→      case 'image': return <span className="text-lg">🖼️</span>;
   598→      case 'video': return <span className="text-lg">🎬</span>;
   599→      case 'audio': return <span className="text-lg">🎵</span>;
   600→      case 'pdf': return <FileText className="h-5 w-5 text-red-500" />;
   601→      case 'text': return <FileText className="h-5 w-5 text-gray-500" />;
   602→      case 'code': return <FileText className="h-5 w-5 text-emerald-500" />;
   603→      case 'docx': return <FileText className="h-5 w-5 text-blue-500" />;
   604→      case 'xlsx': return <FileSpreadsheet className="h-5 w-5 text-emerald-500" />;
   605→      case 'pptx': return <Presentation className="h-5 w-5 text-orange-500" />;
   606→      default: return <FileText className="h-5 w-5 text-muted-foreground" />;
   607→    }
   608→  };
   609→
   610→  // ---- Error rendering ----
   611→  if (error) {
   612→    return (
   613→      <div className="min-h-screen flex flex-col bg-background">
   614→        {/* MODUL 65: Header reuses dark-theme tokens from app-shell */}
   615→        <div className="sticky top-0 z-30 flex items-center justify-between px-4 py-3 border-b border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60">
   616→          <div className="flex items-center gap-3">
   617→            <Button variant="ghost" size="icon" onClick={handleClose} className="text-foreground"><ArrowLeft className="h-5 w-5" /></Button>
   618→            <span className="font-medium truncate max-w-[50vw] text-foreground">{name}</span>
   619→          </div>
   620→          <div className="flex items-center gap-2">
   621→            {supportsHiFi && (
   622→              <Button variant="outline" size="sm" onClick={handleHiFiToggle}>
   623→                {hiFiMode ? <><Zap className="h-4 w-4 mr-2" />Tampilan Cepat</> : <><Eye className="h-4 w-4 mr-2" />Tampilan Asli (PDF)</>}
   624→              </Button>
   625→            )}
   626→            <Button variant="outline" size="sm" asChild>
   627→              <a href={downloadUrl} download={name}><Download className="h-4 w-4 mr-2" />Download</a>
   628→            </Button>
   629→          </div>
   630→        </div>
   631→        <div className="flex-1 flex items-center justify-center p-6">
   632→          <Card className="w-full max-w-md">
   633→            <CardContent className="p-6 flex flex-col items-center gap-4">
   634→              <AlertTriangle className="h-8 w-8 text-destructive" />
   635→              <p className="text-sm text-muted-foreground text-center">{error}</p>
   636→              <div className="flex items-center gap-2">
   637→                {supportsHiFi && (
   638→                  <Button variant="outline" size="sm" onClick={handleHiFiToggle}>
   639→                    {hiFiMode ? <><Zap className="h-4 w-4 mr-2" />Tampilan Cepat</> : <><Eye className="h-4 w-4 mr-2" />Tampilan Asli (PDF)</>}
   640→                  </Button>
   641→                )}
   642→                <Button variant="outline" size="sm" asChild>
   643→                  <a href={downloadUrl} download={name}><Download className="h-4 w-4 mr-2" />Download</a>
   644→                </Button>
   645→              </div>
   646→            </CardContent>
   647→          </Card>
   648→        </div>
   649→      </div>
   650→    );
   651→  }
   652→
   653→  // ---- Main rendering ----
   654→  return (
   655→    <div className="min-h-screen flex flex-col bg-background">
   656→      {/* MODUL 65: Toolbar — reuses dark-theme tokens from app-shell */}
   657→      {/* 65.5: z-30 ensures header stays above canvas-rendered content in both modes */}
   658→      <div className="sticky top-0 z-30 flex items-center justify-between px-4 py-3 border-b border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60">
   659→        {/* 65.4: Explicit gap-3 between element groups */}
   660→        <div className="flex items-center gap-3 min-w-0">
   661→          <Button variant="ghost" size="icon" onClick={handleClose} className="text-foreground shrink-0"><ArrowLeft className="h-5 w-5" /></Button>
   662→          <span className="shrink-0">{getIcon()}</span>
   663→          {/* 65.3: File name with proper foreground contrast */}
   664→          <span className="font-medium truncate max-w-[40vw] text-foreground">{name}</span>
   665→          {/* 65.4: Size badge with explicit spacing */}
   666→          {sizeBytes > 0 && (
   667→            <span className="text-xs text-muted-foreground bg-muted px-2 py-0.5 rounded shrink-0">{formatFileSize(sizeBytes)}</span>
   668→          )}
   669→          {/* 65.4: Type badge with explicit spacing */}
   670→          <span className="text-xs text-muted-foreground bg-muted px-2 py-0.5 rounded shrink-0">{mimeLabel}</span>
   671→          {/* Hi-Fi mode indicator */}
   672→          {hiFiMode && supportsHiFi && (
   673→            <span className="text-xs bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300 px-2 py-0.5 rounded-full font-medium shrink-0">
   674→              Tampilan Asli
   675→            </span>
   676→          )}
   677→        </div>
   678→        {/* 65.4: Right side with explicit gap */}
   679→        <div className="flex items-center gap-2 shrink-0">
   680→          {/* MODUL 64.3: Toggle between High-Fidelity (default) and Fast (secondary) */}
   681→          {supportsHiFi && (
   682→            <Button
   683→              variant={hiFiMode ? 'default' : 'outline'}
   684→              size="sm"
   685→              onClick={handleHiFiToggle}
   686→              title={hiFiMode ? 'Beralih ke tampilan cepat (ringan, tanpa konversi)' : 'Beralih ke tampilan asli (PDF) — presisi tinggi via LibreOffice'}
   687→            >
   688→              {hiFiMode ? (
   689→                <><Zap className="h-4 w-4 mr-2" />Tampilan Cepat</>
   690→              ) : (
   691→                <><Eye className="h-4 w-4 mr-2" />Tampilan Asli (PDF)</>
   692→              )}
   693→            </Button>
   694→          )}
   695→          <Button variant="outline" size="sm" asChild>
   696→            <a href={downloadUrl} download={name}><Download className="h-4 w-4 mr-2" />Download</a>
   697→          </Button>
   698→        </div>
   699→      </div>
   700→
   701→      {/* Content area */}
   702→      <div className="flex-1 overflow-auto">
   703→        {/* ---- MODUL 64.5: High-Fidelity Mode with explicit loading for cache-miss ---- */}
   704→        {hiFiMode && supportsHiFi ? (
   705→          <div className="flex items-center justify-center p-6">
   706→            <PdfPreview contentUrl={convertPdfUrl} />
   707→          </div>
   708→        ) : loading ? (
   709→          /* ---- Loading state ---- */
   710→          <div className="flex items-center justify-center py-20">
   711→            <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   712→            <span className="ml-3 text-sm text-muted-foreground">Loading…</span>
   713→          </div>
   714→        ) : previewType === 'image' ? (
   715→          /* Image — centered, max-width A4 style */
   716→          <div className="flex items-center justify-center p-6 min-h-[80vh] bg-muted/30">
   717→            <img
   718→              src={contentUrl}
   719→              alt={name}
   720→              className="max-w-[210mm] max-h-[80vh] object-contain rounded-lg shadow-md bg-white dark:bg-neutral-800"
   721→            />
   722→          </div>
   723→        ) : previewType === 'video' ? (
   724→          /* Video — centered player */
   725→          <div className="flex items-center justify-center p-6 min-h-[80vh]">
   726→            <video
   727→              src={contentUrl}
   728→              controls
   729→              autoPlay
   730→              className="max-w-[210mm] rounded-lg shadow-md"
   731→              style={{ maxHeight: '80vh' }}
   732→            >
   733→              Your browser does not support video playback.
   734→            </video>
   735→          </div>
   736→        ) : previewType === 'audio' ? (
   737→          /* Audio — centered player with visual wrapper */
   738→          <div className="flex flex-col items-center justify-center p-6 min-h-[80vh]">
   739→            <div className="w-full max-w-[210mm] bg-card rounded-lg shadow-md p-8 flex flex-col items-center gap-6">
   740→              <span className="text-4xl">🎵</span>
   741→              <h2 className="text-lg font-medium text-card-foreground">{name}</h2>
   742→              <audio src={contentUrl} controls autoPlay className="w-full" />
   743→            </div>
   744→          </div>
   745→        ) : previewType === 'pdf' ? (
   746→          /* PDF — canvas rendering with zoom controls */
   747→          <div className="flex items-center justify-center p-6">
   748→            <PdfPreview contentUrl={contentUrl} />
   749→          </div>
   750→        ) : previewType === 'text' || previewType === 'code' ? (
   751→          /* Text/Code — A4-style document view */
   752→          <div className="flex justify-center p-6">
   753→            <div className="w-full max-w-[210mm] min-h-[297mm] bg-card rounded-lg shadow-md p-[20mm] font-mono text-sm leading-relaxed overflow-auto whitespace-pre-wrap break-words border text-card-foreground">
   754→              {textContent}
   755→            </div>
   756→          </div>
   757→        ) : previewType === 'docx' ? (
   758→          /* DOCX — docx-preview (primary) or mammoth (fallback) */
   759→          docxRenderMethod === 'mammoth' && mammothHtml ? (
   760→            <div className="flex justify-center p-6">
   761→              <div
   762→                className="w-full max-w-[210mm] min-h-[297mm] bg-card rounded-lg shadow-md p-[20mm] overflow-auto border text-card-foreground"
   763→                dangerouslySetInnerHTML={{ __html: mammothHtml }}
   764→              />
   765→            </div>
   766→          ) : (
   767→            <div className="flex justify-center p-6">
   768→              <div className="docx-preview-wrapper w-full max-w-[210mm] min-h-[297mm] bg-card rounded-lg shadow-md overflow-auto border text-card-foreground">
   769→                <div ref={docxContainerRef} className="w-full" />
   770→              </div>
   771→            </div>
   772→          )
   773→        ) : previewType === 'xlsx' ? (
   774→          /* XLSX — SheetJS with multi-sheet tab switcher */
   775→          spreadsheetData ? (
   776→            <div className="p-6 max-w-full mx-auto">
   777→              <SpreadsheetPreview data={spreadsheetData} name={name} />
   778→            </div>
   779→          ) : null
   780→        ) : previewType === 'pptx' ? (
   781→          /* PPTX — server-side extracted text */
   782→          presentationData ? (
   783→            <div className="p-6 max-w-[210mm] mx-auto">
   784→              <PresentationPreview data={presentationData} />
   785→            </div>
   786→          ) : null
   787→        ) : (
   788→          /* Unknown/unsupported type — offer download */
   789→          <div className="flex flex-col items-center justify-center p-6 min-h-[80vh]">
   790→            <Card className="w-full max-w-md">
   791→              <CardContent className="p-6 flex flex-col items-center gap-4">
   792→                {getIcon()}
   793→                <h3 className="font-medium">{name}</h3>
   794→                <p className="text-sm text-muted-foreground">
   795→                  This file type ({mimeLabel}) cannot be previewed directly.
   796→                </p>
   797→                <Button variant="default" size="sm" asChild>
   798→                  <a href={downloadUrl} download={name}><Download className="h-4 w-4 mr-2" />Download to View</a>
   799→                </Button>
   800→              </CardContent>
   801→            </Card>
   802→          </div>
   803→        )}
   804→      </div>
   805→    </div>
   806→  );
   807→}
