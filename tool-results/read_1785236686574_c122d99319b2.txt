    84→    84→function SpreadsheetPreview({ data, name }: {
    85→    85→  data: {

    87→    87→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;

    92→    92→  const sheetData = data.sheets[activeSheet];
    93→    93→  const maxRows = 100;
    94→    94→
    95→    95→  if (!sheetData) {
    96→    96→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    97→    97→  }
    98→    98→
    99→    99→  return (
   100→   100→    <div className="flex flex-col w-full">
   101→   101→      {/* Sheet tabs */}

   104→   104→          {data.sheetNames.map((sheetName) => (
   105→   105→            <Button
   106→   106→              key={sheetName}
   107→   107→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   108→   108→              size="sm"
   109→   109→              className="text-xs whitespace-nowrap"

   111→   111→            >
   112→   112→              {sheetName}
   113→   113→            </Button>
   114→   114→          ))}
   115→   115→        </div>
   116→   116→      )}
   117→   117→
   118→   118→      {/* Table */}
   119→   119→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   120→   120→        <table className="w-full text-sm">
   121→   121→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   122→   122→            <tr>
   123→   123→              {sheetData.headers.map((header, i) => (
   124→   124→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   125→   125→                  {header || `Column ${i + 1}`}

   127→   127→              ))}
   128→   128→            </tr>
   129→   129→          </thead>

   131→   131→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   132→   132→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   133→   133→                {sheetData.headers.map((header, j) => (
   134→   134→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   135→   135→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   136→   136→                  </td>
   137→   137→                ))}
   138→   138→              </tr>
   139→   139→            ))}
   140→   140→          </tbody>
   141→   141→        </table>
   142→   142→        {sheetData.rows.length > maxRows && (
   143→   143→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   144→   144→            Showing {maxRows} of {sheetData.rows.length} rows
   145→   145→          </div>
   146→   146→        )}
   147→   147→        {sheetData.rows.length === 0 && (
   148→   148→          <div className="p-6 text-center text-sm text-muted-foreground">
   149→   149→            This sheet is empty
   150→   150→          </div>
   151→   151→        )}

   224→   224→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   225→   225→  contentUrl: string;
   226→   226→  name: string;
   227→   227→  sizeBytes?: number;
   228→   228→  mimeLabel: string;
   229→   229→  closeButton: React.ReactNode;
   230→   230→}) {
   231→   231→  const [loading, setLoading] = useState(true);
   232→   232→  const [error, setError] = useState(false);
   233→   233→  const [numPages, setNumPages] = useState(0);
   234→   234→  const [currentPage, setCurrentPage] = useState(1);
   235→   235→  const [scale, setScale] = useState(1.2);
   236→   236→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   237→   237→  const containerRef = useRef<HTMLDivElement>(null);
   238→   238→
   239→   239→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   240→   240→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   241→   241→    const page = await pdfDocument.getPage(pageNum);
   242→   242→    const viewport = page.getViewport({ scale });
   243→   243→
   244→   244→    let canvas = canvasRefs.current.get(pageNum);
   245→   245→    if (!canvas) {
   246→   246→      canvas = document.createElement('canvas');
   247→   247→      canvasRefs.current.set(pageNum, canvas);
   248→   248→      if (containerRef.current) {
   249→   249→        containerRef.current.appendChild(canvas);
   250→   250→      }
   251→   251→    }
   252→   252→
   253→   253→    canvas.width = viewport.width;
   254→   254→    canvas.height = viewport.height;
   255→   255→    canvas.style.width = `${viewport.width}px`;
   256→   256→    canvas.style.height = `${viewport.height}px`;
   257→   257→    canvas.style.marginBottom = '8px';
   258→   258→    canvas.style.display = 'block';
   259→   259→
   260→   260→    const context = canvas.getContext('2d');
   261→   261→    if (!context) return;
   262→   262→
   263→   263→    await page.render({
   264→   264→      canvasContext: context,
   265→   265→      viewport,
   266→   266→    });
   267→   267→  }, []);
   268→   268→
   269→   269→  useEffect(() => {
   270→   270→    let cancelled = false;
   271→   271→
   272→   272→    const loadPdf = async () => {
   273→   273→      try {
   274→   274→        // Dynamic import of pdfjs-dist
   275→   275→        const pdfjsLib = await import('pdfjs-dist');
   276→   276→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   277→   277→
   278→   278→        // Fetch PDF as ArrayBuffer from contentUrl
   279→   279→        const res = await fetch(contentUrl);
   280→   280→        if (!res.ok) throw new Error('Failed to fetch PDF');
   281→   281→        const arrayBuffer = await res.arrayBuffer();
   282→   282→
   283→   283→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   284→   284→        if (!cancelled) {
   285→   285→          setNumPages(pdfDoc.numPages);
   286→   286→          setLoading(false);
   287→   287→        }
   288→   288→
   289→   289→        // Render all pages
   290→   290→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   291→   291→          if (cancelled) break;
   292→   292→          await renderPage(i, pdfDoc, scale);
   293→   293→        }
   294→   294→      } catch (err) {
   295→   295→        if (!cancelled) {
   296→   296→          console.error('PDF preview error:', err);
   297→   297→          setError(true);
   298→   298→          setLoading(false);
   299→   299→        }
   300→   300→      }
   301→   301→    };
   302→   302→
   303→   303→    loadPdf();
   304→   304→    return () => { cancelled = true; };
   305→   305→  }, [contentUrl, renderPage]);
   306→   306→
   307→   307→  // Re-render pages when scale changes
   308→   308→  useEffect(() => {
   309→   309→    if (numPages === 0 || loading) return;
   310→   310→    let cancelled = false;
   311→   311→
   312→   312→    const rerender = async () => {
   313→   313→      try {
   314→   314→        const pdfjsLib = await import('pdfjs-dist');
   315→   315→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   316→   316→
   317→   317→        const res = await fetch(contentUrl);
   318→   318→        if (!res.ok) return;
   319→   319→        const arrayBuffer = await res.arrayBuffer();
   320→   320→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   321→   321→
   322→   322→        // Clear existing canvases
   323→   323→        if (containerRef.current) {
   324→   324→          containerRef.current.innerHTML = '';
   325→   325→        }
   326→   326→        canvasRefs.current.clear();
   327→   327→
   328→   328→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   329→   329→          if (cancelled) break;
   330→   330→          await renderPage(i, pdfDoc, scale);
   331→   331→        }
   332→   332→      } catch {
   333→   333→        // Silently fail on re-render
   334→   334→      }
   335→   335→    };
   336→   336→
   337→   337→    rerender();
   338→   338→    return () => { cancelled = true; };
   339→   339→  }, [scale, numPages, contentUrl, renderPage]);
   340→   340→
   341→   341→  if (error) {
   342→   342→    return (
   343→   343→      <div className="relative flex flex-col w-full">
   344→   344→        {closeButton}
   345→   345→        <Card className="w-full">
   346→   346→          <CardContent className="p-6 flex flex-col items-center gap-4">
   347→   347→            <FileText className="h-12 w-12 text-muted-foreground" />
   348→   348→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   349→   349→            <Button variant="outline" size="sm" asChild>
   350→   350→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   351→   351→                <ExternalLink className="h-4 w-4 mr-2" />
   352→   352→                Open PDF directly
   353→   353→              </a>
   354→   354→            </Button>
   355→   355→          </CardContent>
   356→   356→        </Card>
   357→   357→      </div>
   358→   358→    );
   359→   359→  }
   360→   360→
   361→   361→  return (
   362→   362→    <div className="relative flex flex-col w-full">
   363→   363→      {closeButton}
   364→   364→      <div className="flex items-center gap-2 mb-3">
   365→   365→        <FileText className="h-5 w-5 text-red-500" />
   366→   366→        <span className="font-medium truncate max-w-xs">{name}</span>
   367→   367→        {sizeBytes && (
   368→   368→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   369→   369→        )}
   370→   370→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   371→   371→        <OfflineBadge />
   372→   372→        <div className="flex items-center gap-1 ml-auto">
   373→   373→          {numPages > 1 && (
   374→   374→            <>
   375→   375→              <Button
   376→   376→                variant="outline"
   377→   377→                size="icon"
   378→   378→                className="h-7 w-7"
   379→   379→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   380→   380→                disabled={currentPage === 1}
   381→   381→                aria-label="Previous page"
   382→   382→              >
   383→   383→                <ChevronLeft className="h-3 w-3" />
   384→   384→              </Button>
   385→   385→              <span className="text-xs text-muted-foreground">
   386→   386→                Page {currentPage}/{numPages}
   387→   387→              </span>
   388→   388→              <Button
   389→   389→                variant="outline"
   390→   390→                size="icon"
   391→   391→                className="h-7 w-7"
   392→   392→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   393→   393→                disabled={currentPage === numPages}
   394→   394→                aria-label="Next page"
   395→   395→              >
   396→   396→                <ChevronRight className="h-3 w-3" />
   397→   397→              </Button>
   398→   398→            </>
   399→   399→          )}
   400→   400→          <Button
   401→   401→            variant="outline"
   402→   402→            size="icon"
   403→   403→            className="h-7 w-7"
   404→   404→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   405→   405→            disabled={scale <= 0.5}
   406→   406→            aria-label="Zoom out"
   407→   407→          >
   408→   408→            <ZoomOut className="h-3 w-3" />
   409→   409→          </Button>
   410→   410→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   411→   411→          <Button
   412→   412→            variant="outline"
   413→   413→            size="icon"
   414→   414→            className="h-7 w-7"
   415→   415→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   416→   416→            disabled={scale >= 3}
   417→   417→            aria-label="Zoom in"
   418→   418→          >
   419→   419→            <ZoomIn className="h-3 w-3" />
   420→   420→          </Button>
   421→   421→        </div>
   422→   422→      </div>
   423→   423→
   424→   424→      {loading ? (
   425→   425→        <div className="flex items-center justify-center py-12">
   426→   426→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   427→   427→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   428→   428→        </div>
   429→   429→      ) : (
   430→   430→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   431→   431→          <div
   432→   432→            ref={containerRef}
   433→   433→            className="flex flex-col items-center p-4"
   434→   434→          />
   435→   435→        </ScrollArea>
   436→   436→      )}
   437→   437→    </div>
   438→   438→  );
   439→   439→}

   441→   441→// ============================================================
   442→   442→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   443→   443→// MODUL 51: Added offline message support + cache integration
   444→   444→// ============================================================
   445→   445→function DocxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton, cachedContent, isFromCache, offlineMessage, triggerBackgroundCache, isLoadingCache }: {
   446→   446→  contentUrl: string;
   447→   447→  name: string;
   448→   448→  sizeBytes?: number;
   449→   449→  mimeLabel: string;
   450→   450→  downloadUrl: string;
   451→   451→  closeButton: React.ReactNode;
   452→   452→  cachedContent: string | null;
   453→   453→  isFromCache: boolean;
   454→   454→  offlineMessage: string | null;
   455→   455→  triggerBackgroundCache: (content: string) => Promise<void>;
   456→   456→  isLoadingCache: boolean;
   457→   457→}) {
   458→   458→  const [loading, setLoading] = useState(true);
   459→   459→  const [error, setError] = useState<string | null>(null);
   460→   460→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   461→   461→  const docxContainerRef = useRef<HTMLDivElement>(null);
   462→   462→
   463→   463→  // MODUL 51: If we have cached content and are offline or have a cache match, render it directly
   464→   464→  const shouldUseCache = isFromCache && cachedContent;
   465→   465→
   466→   466→  useEffect(() => {
   467→   467→    // MODUL 51: If we have cached HTML content, use it directly
   468→   468→    if (shouldUseCache && cachedContent) {
   469→   469→      setMammothHtml(cachedContent);
   470→   470→      setLoading(false);
   471→   471→      return;
   472→   472→    }
   473→   473→
   474→   474→    let cancelled = false;
   475→   475→
   476→   476→    const loadDocx = async () => {
   477→   477→      try {
   478→   478→        // Fetch raw bytes from contentUrl
   479→   479→        const res = await fetch(contentUrl);
   480→   480→        if (!res.ok) throw new Error('Failed to fetch document');
   481→   481→        const arrayBuffer = await res.arrayBuffer();
   482→   482→
