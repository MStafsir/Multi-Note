    85→
    86→  // 39 — Onboarding state
    87→  const [welcomeSlidesDismissed, setWelcomeSlidesDismissed] = useState(false);

    89→
    90→  const { data: onboardingData, isLoading: onboardingLoading } = useQuery({

    92→    queryFn: async () => {

    94→      const data = await res.json();
    95→      if (!data.success) {
    96→        // Return default onboarding state instead of undefined
    97→        return {
    98→          welcomeCompleted: false,
    99→          sampleContentLoaded: false,
   100→          checklistProgress: {},
   101→          dismissedAt: null,
   102→          steps: [],
   103→        };
   104→      }
   105→      return data.data as {
   106→        welcomeCompleted: boolean;
   107→        sampleContentLoaded: boolean;
   108→        checklistProgress: Record<string, boolean>;
   109→        dismissedAt: string | null;

   115→
   116→  // 39 — Derived: show welcome slides when onboarding data indicates it's needed

   118→

   140→    onboardingCompleteMutation.mutate({ dismiss: true });

   143→  const handleOnboardingDismiss = useCallback(() => {
   144→    // Checklist was dismissed — no further action needed, state is already updated via mutation

   177→  const searchRef = useRef<HTMLDivElement>(null);
   178→
   179→  // 22 — Helper: check if user is typing in an input field
   180→  const isUserTyping = useCallback((e: KeyboardEvent): boolean => {
   181→    const target = e.target as HTMLElement;
   182→    if (!target) return false;
   183→    const tagName = target.tagName.toLowerCase();
   184→    // Check if target is input, textarea, or contenteditable
   185→    if (tagName === 'input' || tagName === 'textarea') return true;
   186→    if (target.isContentEditable) return true;
   187→    // Also check if target is inside a cmdk input (command palette)
   188→    if (target.closest('[cmdk-input]')) return true;
   189→    return false;
   190→  }, []);
   191→
   192→  // 22 — Global keyboard shortcuts (single useEffect with keydown listener)
   193→  const handleKeyDown = useCallback(
   194→    (e: KeyboardEvent) => {
   195→      // Command Palette: Cmd/Ctrl+K (new shortcut, replaces old calculator shortcut)
   196→      if ((e.metaKey || e.ctrlKey) && !e.shiftKey && e.key === 'k') {
   197→        e.preventDefault();
   198→        setCommandPaletteOpen(true);
   199→        markOnboardingStep('use_command_palette');
   200→        return;
   201→      }
   202→
   203→      // Calculator: Cmd/Ctrl+Shift+K (moved from Ctrl+K)
   204→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'K' || e.key === 'k')) {
   205→        e.preventDefault();
   206→        toggleOpen();
   207→        markOnboardingStep('use_calculator');
   208→        return;
   209→      }
   210→
   211→      // Search: Cmd/Ctrl+Shift+F
   212→      if ((e.ctrlKey || e.metaKey) && e.shiftKey && (e.key === 'F' || e.key === 'f')) {
   213→        e.preventDefault();
   214→        setHeaderSearchOpen(true);
   215→        markOnboardingStep('use_search');
   216→        setTimeout(() => {
   217→          const input = searchRef.current?.querySelector('input');
   218→          if (input) input.focus();
   219→        }, 50);
   220→        return;
   221→      }
   222→
   223→      // Undo: Cmd/Ctrl+Z
   224→      if ((e.metaKey || e.ctrlKey) && e.key === 'z' && !e.shiftKey) {
   225→        e.preventDefault();
   226→        const action = popAction();
   227→        if (action) {
   228→          // Note: actual undo logic would be implemented per action type
   229→          // This just pops from the stack; the real undo needs to be wired per action
   230→          console.log('Undo action:', action.description);
   231→        }
   232→        return;
   233→      }
   234→
   235→      // Single-key shortcuts only work when user is NOT typing
   236→      if (isUserTyping(e)) return;
   237→
   238→      // Only process single-key shortcuts when no dialog/palette is open
   239→      if (commandPaletteOpen) return;
   240→
   241→      // New Note: N key
   242→      if (e.key === 'n' || e.key === 'N') {
   243→        e.preventDefault();
   244→        setPaletteCreateType('note');
   245→        setPaletteCreateDialogOpen(true);
   246→        return;
   247→      }
   248→
   249→      // New Folder: F key
   250→      if (e.key === 'f' || e.key === 'F') {
   251→        e.preventDefault();
   252→        setPaletteCreateType('folder');
   253→        setPaletteCreateDialogOpen(true);
   254→        return;
   255→      }
   256→
   257→      // Delete selected: Delete/Backspace key
   258→      if (e.key === 'Delete' || e.key === 'Backspace') {
   259→        const selectedIds = Array.from(selectedNodeIds);
   260→        if (selectedIds.length > 0) {
   261→          e.preventDefault();
   262→          // Delete the first selected item
   263→          deleteMutation.mutate({ nodeId: selectedIds[0] });
   264→        }
   265→        return;
   266→      }
   267→    },
   268→    [toggleOpen, isUserTyping, popAction, commandPaletteOpen, selectedNodeIds, deleteMutation]
   269→  );
   270→
   271→  useEffect(() => {
   272→    document.addEventListener('keydown', handleKeyDown);
   273→    return () => document.removeEventListener('keydown', handleKeyDown);
   274→  }, [handleKeyDown]);
   275→

   280→    parentId: string | null

   283→    if (nodeType === 'folder') {
   284→      const folderNode = flatNodes.get(nodeId);
   285→      if (folderNode) {
   286→        setCurrentFolder(nodeId, []);
   287→      }
   288→    } else if (nodeType === 'note') {
   289→      if (parentId) {
   290→        setCurrentFolder(parentId, []);
   291→      }
   292→    } else {
   293→      if (parentId) {
   294→        setCurrentFolder(parentId, []);
   295→      }
   296→    }
   297→  }, [flatNodes, setCurrentFolder]);
   298→

   300→  const toggleSidebar = useCallback(() => {
   301→    if (isMobile) {
   302→      setSidebarOpen(prev => !prev);
   303→    } else {
   304→      if (sidebarOpen) {
   305→        setSidebarCollapsed(!sidebarCollapsed);
   306→      } else {
   307→        setSidebarOpen(true);
   308→        setSidebarCollapsed(false);
   309→      }
   310→    }
   311→  }, [isMobile, sidebarOpen, sidebarCollapsed]);
   312→

   314→  const closeSidebar = useCallback(() => {
   315→    setSidebarOpen(false);
   316→  }, []);
   317→

   319→  const sheetY = useMotionValue(0);
   320→  const sheetOpacity = useTransform(sheetY, [0, 200], [1, 0]);
   321→
   322→  // Handle pan gesture on bottom sheet
   323→  const handleSheetPan = useCallback((event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
   324→    if (info.offset.y > 80) {
   325→      closeSidebar();
   326→    }
   327→  }, [closeSidebar]);
   328→
   329→
   330→
   331→  return (
   332→    <WorkspaceDndProvider>
   333→      <TooltipProvider>

   336→          <a
   337→            href="#main-content"
   338→            className="sr-only focus:not-sr-only focus:fixed focus:top-2 focus:left-2 focus:z-[100] focus:bg-primary focus:text-primary-foreground focus:px-4 focus:py-2 focus:rounded-md focus:text-sm focus:font-medium focus:shadow-lg"
   339→          >
   340→            Skip to main content
   341→          </a>

   395→                    className="md:hidden min-h-[44px] min-w-[44px]"
   396→                    onClick={() => setCommandPaletteOpen(true)}
   397→                    aria-label="Open command palette"
   398→                  >
   399→                    <Search className="h-5 w-5" />
   400→                  </Button>
   401→                )}
   402→              </div>
   403→
   404→              <div className="flex items-center gap-3">
   405→                {/* MODUL 47 — Locale switcher */}
   406→                <LocaleSwitcher />
   407→
   408→                {/* Notification badge */}
   409→                <NotificationBadge />
   410→
   411→                {/* Calculator toggle button — updated shortcut tooltip */}
   412→                <Tooltip>
   413→                  <TooltipTrigger asChild>
   414→                    <Button
   415→                      variant={isOpen ? 'secondary' : 'ghost'}
   416→                      size="icon"
   417→                      className="h-9 w-9 shrink-0 min-h-[44px] min-w-[44px]"
   418→                      onClick={toggleOpen}
   419→                      aria-label="Toggle calculator (Ctrl+Shift+K)"
   420→                    >
   421→                      <Calculator className={`h-4 w-4 ${isOpen ? 'text-orange-500' : 'text-muted-foreground'}`} />
   422→                    </Button>

   429→                {/* User menu dropdown */}

   433→                      <div className="flex items-center justify-center w-8 h-8 rounded-full bg-neutral-200 dark:bg-neutral-800 text-neutral-700 dark:text-neutral-300 text-sm font-medium">

   437→                    </Button>

   439→                  <DropdownMenuContent align="end" className="w-56">

   441→                      <div className="flex flex-col">
   442→                        <span className="font-medium">{user?.name || 'User'}</span>
   443→                        <span className="text-xs text-muted-foreground">{user?.email}</span>
   444→                      </div>
   445→                    </DropdownMenuLabel>
   446→                    <DropdownMenuSeparator />
   447→                    {currentWorkspaceId && (currentWorkspaceRole === 'owner' || currentWorkspaceRole === 'admin') && (
   448→                      <DropdownMenuItem
   449→                        onClick={() => setWorkspaceSettingsOpen(true)}
   450→                      >
   451→                        <Building2 className="h-4 w-4 mr-2" />
   452→                        Workspace Settings
   453→                      </DropdownMenuItem>
   454→                    )}
   455→                    <DropdownMenuItem
   456→                      onClick={() => setInvitationViewOpen(true)}
   457→                    >
   458→                      <Mail className="h-4 w-4 mr-2" />
   459→                      Invitations
   460→                    </DropdownMenuItem>
   461→                    <DropdownMenuItem
   462→                      onClick={() => setSettingsDialogOpen(true)}
   463→                    >
   464→                      <Settings className="h-4 w-4 mr-2" />
   465→                      Settings
   466→                    </DropdownMenuItem>
   467→                    <DropdownMenuSeparator />

   469→                      onClick={() => {
   470→                        useAuthStore.getState().logout();
   471→                        signOut({ redirect: false });
   472→                      }}
   473→                      className="text-destructive focus:text-destructive"
   474→                    >
   475→                      <LogOut className="h-4 w-4 mr-2" />
   476→                      Sign Out
   477→                    </DropdownMenuItem>
   478→                  </DropdownMenuContent>
   479→                </DropdownMenu>
   480→              </div>
   481→            </div>
   482→          </header>
   483→
   484→          {/* Main content: sidebar + workspace */}
   485→          <div className="flex-1 flex overflow-hidden relative">
   486→            {/* 29 — aria-live region for toast announcements */}

   488→
   489→            {/* ===== Mobile: Bottom-sheet drawer pattern ===== */}
   490→            {isMobile && (
   491→              <AnimatePresence initial={false}>
   492→                {sidebarOpen && (
   493→                  <>
   494→                    {/* Backdrop */}
   495→                    <motion.div
   496→                      initial={{ opacity: 0 }}
   497→                      animate={{ opacity: 1 }}
   498→                      exit={{ opacity: 0 }}
   499→                      transition={{ duration: 0.2 }}
   500→                      className="fixed inset-0 bg-black/40 z-20"
   501→                      onClick={closeSidebar}
   502→                      role="button"
   503→                      tabIndex={-1}
   504→                      aria-label="Close sidebar overlay"
   505→                    />
   506→
   507→                    {/* Bottom sheet */}
   508→                    <motion.aside
   509→                      role="complementary"
   510→                      aria-label="Sidebar navigation"
   511→                      initial={{ y: '100%' }}
   512→                      animate={{ y: 0 }}
   513→                      exit={{ y: '100%' }}
   514→                      transition={{ type: 'spring', damping: 30, stiffness: 300 }}
   515→                      drag="y"
   516→                      dragConstraints={{ top: 0 }}
   517→                      dragElastic={0.2}
   518→                      onDragEnd={handleSheetPan}
   519→                      style={{ y: sheetY, opacity: sheetOpacity }}
   520→                      className="fixed bottom-0 left-0 right-0 z-20 bg-sidebar border-t border-border rounded-t-2xl overflow-hidden shadow-xl"
   521→                    >
   522→                      {/* Drag handle bar */}
   523→                      <div className="flex items-center justify-center pt-3 pb-2 cursor-grab active:cursor-grabbing">
   524→                        <div className="w-12 h-1.5 rounded-full bg-neutral-300 dark:bg-neutral-700" />
   525→                      </div>
   526→
   527→                      {/* Close button */}
