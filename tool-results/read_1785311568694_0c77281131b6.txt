   130→  // since nodeId varies per item click

   132→    const node = flatNodes.get(nodeId);
   133→    if (!node) return;
   134→    const res = await fetch(`/api/nodes/${nodeId}/favorite`, {
   135→      method: 'PATCH',
   136→      headers: { 'Content-Type': 'application/json' },
   137→      body: JSON.stringify({ isFavorite: !node.isFavorite }),
   138→    });

   141→      // Optimistic update: toggle isFavorite in the local tree
   142→      const flatNodesUpdate = new Map(flatNodes);
   143→      const existing = flatNodesUpdate.get(nodeId);
   144→      if (existing) {
   145→        flatNodesUpdate.set(nodeId, { ...existing, isFavorite: !existing.isFavorite });

   170→  };
   171→

   176→      const newSelection = new Set(multiSelectedIds);

   178→        newSelection.delete(node.id);
   179→      } else {
   180→        newSelection.add(node.id);
   181→      }
   182→      setMultiSelectedIds(newSelection);
   183→      return;
   184→    }
   185→
   186→    // Single click: select the item
   187→    setSelectedNodeId(node.id);
   188→    setMultiSelectedIds(new Set());
   189→  };

   191→  // Double click: open item (folder → navigate, note → editor, file → preview)

   194→    if (node.type === 'folder') {
   195→      navigateToFolder(node.id, node.name);
   196→    } else if (node.type === 'note') {
   197→      openNote(node.id);

   199→      const mimeType = node.metadata?.mimeType || 'application/octet-stream';
   200→      const tier = getPreviewTier(mimeType);
   201→
   202→      if (tier === 'tier1_native') {
   203→        // Tier 1 (image/video/audio/PDF/text) → Mode A: inline overlay/modal

   205→        setPreviewFileName(node.name);
   206→        setPreviewFileMime(mimeType);
   207→        setPreviewFileSize(typeof node.metadata?.sizeBytes === 'number' ? node.metadata.sizeBytes : 0);
   208→        setPreviewFileChecksum(node.metadata?.checksumSha256 || null);
   209→        setPreviewModalOpen(true);
   210→      } else {
   211→        // Tier 2/3 (DOCX/XLSX/PPTX) → Mode B: open dedicated viewer in new tab
   212→        window.open('/view/' + node.id, '_blank');
   213→      }
   214→    }

   216→

   221→    setSelectedNodeId(null);
   222→  };
   223→

   227→      const parentPath = currentFolderPath.slice(0, -1);
   228→      const parentFolder = parentPath[parentPath.length - 1];
   229→      setCurrentFolder(parentFolder.id, parentPath);
   230→      setSelectedNodeId(null);
   231→    }
   232→  };
   233→
   234→  // Navigate via breadcrumb
   235→  const navigateBreadcrumb = (index: number) => {
   236→    const newPath = currentFolderPath.slice(0, index + 1);
   237→    const target = newPath[newPath.length - 1];
   238→    setCurrentFolder(target.id, newPath);
   239→    setSelectedNodeId(null);
   240→  };
   241→
   242→  // Open note editor
   243→  const openNote = (noteId: string) => {
   244→    setSelectedNodeId(noteId);
   245→  };
   246→
   247→  const handleRename = (nodeId: string, currentName: string) => {
   248→    setRenameNodeId(nodeId);
   249→    setRenameNodeName(currentName);
   250→    setRenameDialogOpen(true);
   251→  };
   252→
   253→  const handleDelete = (nodeId: string) => {
   254→    deleteMutation.mutate({ nodeId });
   255→  };
   256→
   257→  const handleShare = (node: TreeNode) => {
   258→    setShareNode(node);
   259→    setShareDialogOpen(true);
   260→    markOnboardingStep('share_item');
   261→  };
   262→
   263→  const handleVersionHistory = (node: TreeNode) => {
   264→    setVersionNodeId(node.id);
   265→    setVersionFileName(node.name);
   266→    setVersionDialogOpen(true);
   267→  };
   268→
   269→  const formatBytes = (bytes: number): string => {
   270→    if (!bytes || bytes < 0) return '—';
   271→    const k = 1024;
   272→    const sizes = ['B', 'KB', 'MB', 'GB'];
   273→    const i = Math.floor(Math.log(bytes) / Math.log(k));
   274→    return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`;
   275→  };
   276→
   277→  const formatDate = (dateStr: string): string => {
   278→    try {
   279→      const date = new Date(dateStr);
   280→      return date.toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' });
   281→    } catch {
   282→      return '—';
   283→    }
   284→  };

   344→              exit={{ width: 0, opacity: 0 }}
   345→              transition={{ duration: 0.2 }}
   346→              className="shrink-0 overflow-hidden"
   347→            >
   348→              <RevisionSidebar
   349→                nodeId={selectedNode.id}
   350→                onClose={() => setShowRevisionSidebar(false)}
   351→              />
   352→            </motion.div>
   353→          )}
   354→        </AnimatePresence>
   355→      </div>

   357→  }

   361→      {/* Toolbar */}
   362→      <div className="sticky top-0 z-10 bg-background/95 backdrop-blur border-b border-border px-6 py-3">
   363→        <div className="flex items-center gap-4">
   364→          {/* Breadcrumb — 29: wrapped in <nav> */}
   365→          <nav aria-label="Breadcrumb">
   366→          <Breadcrumb className="flex-1 min-w-0">

   405→                  navigateToFolder(nodeId, folderNode.name);
   406→                }

   416→              } else {

   419→                  const parentFolder = flatNodes.get(parentId);
   420→                  if (parentFolder) {
   421→                    setCurrentFolder(parentId, []);
   422→                  }
   423→                }
   424→                setSelectedNodeId(nodeId);
   425→              }
   426→            }}
   427→          />
   428→
   429→          {/* View mode toggle */}
   430→          <div className="flex items-center border rounded-md">
   431→            <Button
   432→              variant={viewMode === 'grid' ? 'secondary' : 'ghost'}
   433→              size="icon"
   434→              className="h-8 w-8 min-h-[44px] min-w-[44px]"
   435→              onClick={() => setViewMode('grid')}
   436→              aria-label="Grid view"
   437→            >
   438→              <Grid3X3 className="h-4 w-4" />
   439→            </Button>
   440→            <Button
   441→              variant={viewMode === 'list' ? 'secondary' : 'ghost'}
   442→              size="icon"
   443→              className="h-8 w-8 min-h-[44px] min-w-[44px]"
   444→              onClick={() => setViewMode('list')}
   445→              aria-label="List view"
   446→            >
   447→              <List className="h-4 w-4" />
   448→            </Button>
   449→          </div>
   450→
   451→          {/* Upload button — always visible in toolbar */}
   452→          <Button
   453→            variant="outline"
   454→            size="sm"
   455→            className="min-h-[44px]"
   456→            onClick={() => fileInputEl?.click()}
   457→            aria-label="Upload files to current folder"
   458→          >
   459→            <Upload className="h-4 w-4 mr-1.5" />
   460→            Upload
   461→          </Button>
   462→
   463→          {/* Offline status badge — 51: shows online/offline status */}
   464→          <OfflineBadge />
   465→        </div>
   466→      </div>
   467→
   468→      {/* Bulk Action Toolbar — 18.2: appears when multi-select is active */}
   469→      {multiSelectedIds.size > 0 && (
   470→        <BulkActionToolbar
   471→          selectedIds={multiSelectedIds}
   472→          onClearSelection={() => setMultiSelectedIds(new Set())}

   474→      )}
   475→
   476→      {/* Content */}
   477→      <ScrollArea className="flex-1">
   478→        <div className="p-6">
   479→
   480→          {/* Loading state */}
   481→          {isLoading && (
   482→            <div className="flex items-center justify-center py-12">
   483→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   484→            </div>
   485→          )}
   486→
   487→          {/* 39 — Empty state with onboarding CTAs */}
   488→          {!isLoading && filteredItems.length === 0 && (
   489→            <EmptyStateCTA
   490→              onUploadFile={() => {
   491→                fileInputEl?.click();
   492→              }}
   493→              onCreateNote={() => {
   494→                setCreateType('note');
   495→                setCreateDialogOpen(true);
   496→                markOnboardingStep('create_note');
   497→              }}
   498→              onOpenTemplateGallery={() => {
   499→                setTemplateGalleryOpen(true);
   500→              }}
   501→              parentId={currentFolderId}
   502→            />
   503→          )}
   504→          {/* Hidden file input for upload CTA */}
   505→          <input
   506→            ref={setFileInputEl}
   507→            type="file"
   508→            multiple
   509→            className="hidden"
   510→            aria-hidden="true"
   511→            onChange={(e) => {
   512→              if (e.target.files) {
   513→                for (const file of Array.from(e.target.files)) {
   514→                  uploadMutation.mutate({ file, parentId: currentFolderId });
   515→                }
   516→                markOnboardingStep('upload_file');
   517→              }
   518→              e.target.value = ''; // Reset input
   519→            }}
   520→          />
   521→
   522→          {/* Items */}
   523→          {!isLoading && filteredItems.length > 0 && (
   524→            <AnimatePresence mode="wait">
   525→              <motion.div
   526→                key={viewMode}
   527→                initial={{ opacity: 0 }}
   528→                animate={{ opacity: 1 }}
   529→                exit={{ opacity: 0 }}
   530→                transition={{ duration: 0.15 }}
   531→              >
   532→                {viewMode === 'grid' ? (
   533→                  <ul role="list" aria-label="Folder contents grid" className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
   534→                    {/* + Add New card — always visible at start of every folder */}
   535→                    <li role="listitem" aria-label="Add new item">
   536→                      <DropdownMenu>
   537→                        <DropdownMenuTrigger asChild>
   538→                          <Card className="cursor-pointer hover:border-primary/40 hover:shadow-md transition-all group">
   539→                            <CardContent className="p-4 flex flex-col items-center text-center gap-2">
   540→                              <div className="w-12 h-12 rounded-lg bg-primary/10 flex items-center justify-center group-hover:bg-primary/20 transition-colors">
   541→                                <Plus className="h-6 w-6 text-primary" />
