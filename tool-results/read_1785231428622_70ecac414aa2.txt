    76→// Spreadsheet Preview Sub-component
    77→// Renders xlsx data as a scrollable table with sheet tabs
    78→// ============================================================
    79→function SpreadsheetPreview({ data, name }: {
    80→  data: {
    81→    sheetNames: string[];
    82→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
    83→  };
    84→  name: string;
    85→}) {
    86→  const [activeSheet, setActiveSheet] = useState(data.sheetNames[0] || '');
    87→  const sheetData = data.sheets[activeSheet];
    88→  const maxRows = 100;
    89→
    90→  if (!sheetData) {
    91→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    92→  }
    93→
    94→  return (
    95→    <div className="flex flex-col w-full">
    96→      {/* Sheet tabs */}
    97→      {data.sheetNames.length > 1 && (
    98→        <div className="flex items-center gap-1 mb-3 overflow-x-auto pb-1">
    99→          {data.sheetNames.map((sheetName) => (
   100→            <Button
   101→              key={sheetName}
   102→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   103→              size="sm"
   104→              className="text-xs whitespace-nowrap"
   105→              onClick={() => setActiveSheet(sheetName)}
   106→            >
   107→              {sheetName}
   108→            </Button>
   109→          ))}
   110→        </div>
   111→      )}
   112→
   113→      {/* Table */}
   114→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   115→        <table className="w-full text-sm">
   116→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   117→            <tr>
   118→              {sheetData.headers.map((header, i) => (
   119→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   120→                  {header || `Column ${i + 1}`}
   121→                </th>
   122→              ))}
   123→            </tr>
   124→          </thead>
   125→          <tbody>
   126→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   127→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   128→                {sheetData.headers.map((header, j) => (
   129→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   130→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   131→                  </td>
   132→                ))}
   133→              </tr>
   134→            ))}
   135→          </tbody>
   136→        </table>
   137→        {sheetData.rows.length > maxRows && (
   138→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   139→            Showing {maxRows} of {sheetData.rows.length} rows
   140→          </div>
   141→        )}
   142→        {sheetData.rows.length === 0 && (
   143→          <div className="p-6 text-center text-sm text-muted-foreground">
   144→            This sheet is empty
   145→          </div>
   146→        )}
   147→      </div>
   148→    </div>
   149→  );
   150→}

   217→// PDF Preview (Tier 1 — pdfjs-dist canvas rendering)

   219→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   220→  contentUrl: string;
   221→  name: string;
   222→  sizeBytes?: number;
   223→  mimeLabel: string;
   224→  closeButton: React.ReactNode;
   225→}) {
   226→  const [loading, setLoading] = useState(true);
   227→  const [error, setError] = useState(false);
   228→  const [numPages, setNumPages] = useState(0);
   229→  const [currentPage, setCurrentPage] = useState(1);
   230→  const [scale, setScale] = useState(1.2);
   231→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   232→  const containerRef = useRef<HTMLDivElement>(null);
   233→
   234→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   235→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   236→    const page = await pdfDocument.getPage(pageNum);
   237→    const viewport = page.getViewport({ scale });
   238→
   239→    let canvas = canvasRefs.current.get(pageNum);
   240→    if (!canvas) {
   241→      canvas = document.createElement('canvas');
   242→      canvasRefs.current.set(pageNum, canvas);
   243→      if (containerRef.current) {
   244→        containerRef.current.appendChild(canvas);
   245→      }
   246→    }
   247→
   248→    canvas.width = viewport.width;
   249→    canvas.height = viewport.height;
   250→    canvas.style.width = `${viewport.width}px`;
   251→    canvas.style.height = `${viewport.height}px`;
   252→    canvas.style.marginBottom = '8px';
   253→    canvas.style.display = 'block';
   254→
   255→    const context = canvas.getContext('2d');
   256→    if (!context) return;
   257→
   258→    await page.render({
   259→      canvasContext: context,
   260→      viewport,
   261→    });
   262→  }, []);
   263→
   264→  useEffect(() => {
   265→    let cancelled = false;
   266→
   267→    const loadPdf = async () => {
   268→      try {
   269→        // Dynamic import of pdfjs-dist
   270→        const pdfjsLib = await import('pdfjs-dist');
   271→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   272→
   273→        // Fetch PDF as ArrayBuffer from contentUrl
   274→        const res = await fetch(contentUrl);
   275→        if (!res.ok) throw new Error('Failed to fetch PDF');
   276→        const arrayBuffer = await res.arrayBuffer();
   277→
   278→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   279→        if (!cancelled) {
   280→          setNumPages(pdfDoc.numPages);
   281→          setLoading(false);
   282→        }
   283→
   284→        // Render all pages
   285→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   286→          if (cancelled) break;
   287→          await renderPage(i, pdfDoc, scale);
   288→        }
   289→      } catch (err) {
   290→        if (!cancelled) {
   291→          console.error('PDF preview error:', err);
   292→          setError(true);
   293→          setLoading(false);
   294→        }
   295→      }
   296→    };
   297→
   298→    loadPdf();
   299→    return () => { cancelled = true; };
   300→  }, [contentUrl, renderPage]);
   301→
   302→  // Re-render pages when scale changes
   303→  useEffect(() => {
   304→    if (numPages === 0 || loading) return;
   305→    let cancelled = false;
   306→
   307→    const rerender = async () => {
   308→      try {
   309→        const pdfjsLib = await import('pdfjs-dist');
   310→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   311→
   312→        const res = await fetch(contentUrl);
   313→        if (!res.ok) return;
   314→        const arrayBuffer = await res.arrayBuffer();
   315→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   316→
   317→        // Clear existing canvases
   318→        if (containerRef.current) {
   319→          containerRef.current.innerHTML = '';
   320→        }
   321→        canvasRefs.current.clear();
   322→
   323→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   324→          if (cancelled) break;
   325→          await renderPage(i, pdfDoc, scale);
   326→        }
   327→      } catch {
   328→        // Silently fail on re-render
   329→      }
   330→    };
   331→
   332→    rerender();
   333→    return () => { cancelled = true; };
   334→  }, [scale, numPages, contentUrl, renderPage]);
   335→
   336→  if (error) {
   337→    return (
   338→      <div className="relative flex flex-col w-full">
   339→        {closeButton}
   340→        <Card className="w-full">
   341→          <CardContent className="p-6 flex flex-col items-center gap-4">
   342→            <FileText className="h-12 w-12 text-muted-foreground" />
   343→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   344→            <Button variant="outline" size="sm" asChild>
   345→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   346→                <ExternalLink className="h-4 w-4 mr-2" />
   347→                Open PDF directly
   348→              </a>
   349→            </Button>
   350→          </CardContent>
   351→        </Card>
   352→      </div>
   353→    );
   354→  }
   355→
   356→  return (
   357→    <div className="relative flex flex-col w-full">
   358→      {closeButton}
   359→      <div className="flex items-center gap-2 mb-3">
   360→        <FileText className="h-5 w-5 text-red-500" />
   361→        <span className="font-medium truncate max-w-xs">{name}</span>
   362→        {sizeBytes && (
   363→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   364→        )}
   365→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   366→        <OfflineBadge />
   367→        <div className="flex items-center gap-1 ml-auto">
   368→          {numPages > 1 && (
   369→            <>
   370→              <Button
   371→                variant="outline"
   372→                size="icon"
   373→                className="h-7 w-7"
   374→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   375→                disabled={currentPage === 1}
   376→                aria-label="Previous page"
   377→              >
   378→                <ChevronLeft className="h-3 w-3" />
   379→              </Button>
   380→              <span className="text-xs text-muted-foreground">
   381→                Page {currentPage}/{numPages}
   382→              </span>
   383→              <Button
   384→                variant="outline"
   385→                size="icon"
   386→                className="h-7 w-7"
   387→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   388→                disabled={currentPage === numPages}
   389→                aria-label="Next page"
   390→              >
   391→                <ChevronRight className="h-3 w-3" />
   392→              </Button>
   393→            </>
   394→          )}
   395→          <Button
   396→            variant="outline"
   397→            size="icon"
   398→            className="h-7 w-7"
   399→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   400→            disabled={scale <= 0.5}
   401→            aria-label="Zoom out"
   402→          >
   403→            <ZoomOut className="h-3 w-3" />
   404→          </Button>
   405→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   406→          <Button
   407→            variant="outline"
   408→            size="icon"
   409→            className="h-7 w-7"
   410→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   411→            disabled={scale >= 3}
   412→            aria-label="Zoom in"
   413→          >
   414→            <ZoomIn className="h-3 w-3" />
   415→          </Button>
   416→        </div>
   417→      </div>
   418→
   419→      {loading ? (
   420→        <div className="flex items-center justify-center py-12">
   421→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   422→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   423→        </div>
   424→      ) : (
   425→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   426→          <div
   427→            ref={containerRef}
   428→            className="flex flex-col items-center p-4"
   429→          />
   430→        </ScrollArea>
   431→      )}
   432→    </div>
   433→  );
   434→}

   436→// ============================================================
   437→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   438→// ============================================================
   439→function DocxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton }: {
   440→  contentUrl: string;
   441→  name: string;
   442→  sizeBytes?: number;
   443→  mimeLabel: string;
   444→  downloadUrl: string;
   445→  closeButton: React.ReactNode;
   446→}) {
   447→  const [loading, setLoading] = useState(true);
   448→  const [error, setError] = useState<string | null>(null);
   449→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   450→  const docxContainerRef = useRef<HTMLDivElement>(null);
   451→
   452→  useEffect(() => {
   453→    let cancelled = false;
   454→
   455→    const loadDocx = async () => {
   456→      try {
   457→        // Fetch raw bytes from contentUrl
   458→        const res = await fetch(contentUrl);
   459→        if (!res.ok) throw new Error('Failed to fetch document');
   460→        const arrayBuffer = await res.arrayBuffer();
   461→
   462→        // Try docx-preview first
   463→        try {
   464→          const docxPreview = await import('docx-preview');
   465→          if (docxContainerRef.current && !cancelled) {
   466→            await docxPreview.renderAsync(
   467→              arrayBuffer,
   468→              docxContainerRef.current,
   469→              undefined,
   470→              { className: 'docx-preview-wrapper' }
   471→            );
   472→            if (!cancelled) {
   473→              setLoading(false);
   474→            }
   475→            return;
   476→          }
   477→        } catch (docxPreviewErr) {
   478→          console.warn('docx-preview failed, falling back to mammoth:', docxPreviewErr);
   479→        }
   480→
   481→        // Fallback to mammoth
   482→        if (!cancelled) {
   483→          const mammoth = await import('mammoth');
   484→          const result = await mammoth.convertToHtml({ buffer: arrayBuffer });
   485→          if (!cancelled) {
   486→            setMammothHtml(result.value);
   487→            setLoading(false);
   488→          }
   489→        }
   490→      } catch (err) {
   491→        if (!cancelled) {
   492→          console.error('DOCX preview error:', err);
   493→          setError('Failed to load document preview');
   494→          setLoading(false);
   495→        }
   496→      }
   497→    };
   498→
   499→    loadDocx();
   500→    return () => { cancelled = true; };
   501→  }, [contentUrl]);
   502→
   503→  if (error) {
   504→    return (
   505→      <div className="relative flex flex-col w-full">
   506→        {closeButton}
   507→        <Card className="w-full">
   508→          <CardContent className="p-6 flex flex-col items-center gap-4">
   509→            <FileText className="h-12 w-12 text-muted-foreground" />
   510→            <p className="text-sm text-muted-foreground">{error}</p>
   511→            <Button variant="outline" size="sm" asChild>
   512→              <a href={downloadUrl} download={name}>
   513→                <Download className="h-4 w-4 mr-2" />
   514→                Download to open
   515→              </a>
   516→            </Button>
   517→          </CardContent>
   518→        </Card>
   519→      </div>
   520→    );
   521→  }
   522→
   523→  return (
   524→    <div className="relative flex flex-col w-full">
   525→      {closeButton}
   526→      <div className="flex items-center gap-2 mb-3">
   527→        <FileText className="h-5 w-5 text-blue-500" />
   528→        <span className="font-medium truncate max-w-xs">{name}</span>
   529→        {sizeBytes && (
   530→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   531→        )}
   532→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   533→        <OfflineBadge />
   534→        <OpenWithDropdown nodeId={name} mimeType="application/vnd.openxmlformats-officedocument.wordprocessingml.document" fileName={name} />
   535→      </div>
   536→
   537→      {loading ? (
   538→        <div className="flex items-center justify-center py-12">
   539→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   540→          <span className="ml-2 text-sm text-muted-foreground">Loading document…</span>
   541→        </div>
   542→      ) : mammothHtml ? (
   543→        // Mammoth fallback: render HTML in styled div
   544→        <div
   545→          className="w-full rounded-lg border bg-white p-6 overflow-auto max-h-[70vh] prose prose-sm dark:prose-invert"
   546→          dangerouslySetInnerHTML={{ __html: mammothHtml }}
   547→        />
   548→      ) : (
   549→        // docx-preview rendered into container ref
   550→        <div
   551→          ref={docxContainerRef}
   552→          className="w-full rounded-lg border bg-white overflow-auto max-h-[70vh]"
   553→        />
   554→      )}
   555→    </div>
   556→  );
   557→}

   559→// ============================================================
   560→// XLSX Preview (Tier 2 — client-side SheetJS parse)
   561→// ============================================================
   562→function XlsxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton }: {
   563→  contentUrl: string;
   564→  name: string;
   565→  sizeBytes?: number;
   566→  mimeLabel: string;
   567→  downloadUrl: string;
   568→  closeButton: React.ReactNode;
   569→}) {
   570→  const [loading, setLoading] = useState(true);
   571→  const [error, setError] = useState<string | null>(null);
   572→  const [spreadsheetData, setSpreadsheetData] = useState<{
   573→    sheetNames: string[];
   574→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
   575→  } | null>(null);
   576→
   577→  useEffect(() => {
   578→    let cancelled = false;
   579→
   580→    const loadXlsx = async () => {
   581→      try {
   582→        // Fetch raw bytes from contentUrl
   583→        const res = await fetch(contentUrl);
   584→        if (!res.ok) throw new Error('Failed to fetch spreadsheet');
   585→        const arrayBuffer = await res.arrayBuffer();
   586→
   587→        // Dynamic import of SheetJS
   588→        const XLSX = await import('xlsx');
   589→        const workbook = XLSX.read(arrayBuffer, { type: 'array' });
   590→
   591→        const sheetNames = workbook.SheetNames;
   592→        const sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }> = {};
   593→
   594→        for (const sheetName of sheetNames) {
   595→          const worksheet = workbook.Sheets[sheetName];
   596→          if (!worksheet) continue;
   597→
   598→          const jsonData = XLSX.utils.sheet_to_json<Record<string, string | number | boolean | null>>(worksheet, { defval: null });
   599→          const headers = jsonData.length > 0 ? Object.keys(jsonData[0]) : [];
   600→
   601→          sheets[sheetName] = {
   602→            rows: jsonData.slice(0, 100),
   603→            headers,
   604→          };
   605→        }
   606→
   607→        if (!cancelled) {
   608→          setSpreadsheetData({ sheetNames, sheets });
   609→          setLoading(false);
   610→        }
   611→      } catch (err) {
   612→        if (!cancelled) {
   613→          console.error('XLSX preview error:', err);
   614→          setError('Failed to load spreadsheet preview');
   615→          setLoading(false);
   616→        }
   617→      }
   618→    };
   619→
   620→    loadXlsx();
   621→    return () => { cancelled = true; };
   622→  }, [contentUrl]);
   623→
   624→  if (error) {
   625→    return (
   626→      <div className="relative flex flex-col w-full">
   627→        {closeButton}
   628→        <Card className="w-full">
   629→          <CardContent className="p-6 flex flex-col items-center gap-4">
   630→            <FileSpreadsheet className="h-12 w-12 text-muted-foreground" />
   631→            <p className="text-sm text-muted-foreground">{error}</p>
   632→            <Button variant="outline" size="sm" asChild>
   633→              <a href={downloadUrl} download={name}>
   634→                <Download className="h-4 w-4 mr-2" />
   635→                Download to open
   636→              </a>
   637→            </Button>
   638→          </CardContent>
   639→        </Card>
   640→      </div>
   641→    );
   642→  }
   643→
   644→  return (
   645→    <div className="relative flex flex-col w-full">
   646→      {closeButton}
   647→      <div className="flex items-center gap-2 mb-3">
   648→        <FileSpreadsheet className="h-5 w-5 text-emerald-500" />
   649→        <span className="font-medium truncate max-w-xs">{name}</span>
   650→        {sizeBytes && (
   651→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   652→        )}
   653→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   654→        <OfflineBadge />
   655→        <OpenWithDropdown nodeId={name} mimeType="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" fileName={name} />
   656→      </div>
   657→
   658→      {loading ? (
   659→        <div className="flex items-center justify-center py-12">
   660→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   661→          <span className="ml-2 text-sm text-muted-foreground">Loading spreadsheet…</span>
   662→        </div>
   663→      ) : spreadsheetData ? (
   664→        <SpreadsheetPreview data={spreadsheetData} name={name} />
   665→      ) : null}
   666→    </div>
