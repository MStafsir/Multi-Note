     1→'use client';
     2→
     3→import { Fragment, useState } from 'react';
     4→
     5→import { motion, AnimatePresence } from 'framer-motion';
     6→import {
     7→  Folder,
     8→  File,
     9→  FileText,
    10→  ChevronRight,
    11→  Grid3X3,
    12→  List,
    13→  ArrowUp,
    14→  Loader2,
    15→  MoreHorizontal,
    16→  Pencil,
    17→  Trash2,
    18→  Share2,
    19→  History,
    20→  Star,
    21→} from 'lucide-react';
    22→import type { TreeNode, NodeType } from '@/types';
    23→import { useFileTreeStore } from '@/store/file-tree';
    24→import { Button } from '@/components/ui/button';
    25→
    26→import { Card, CardContent } from '@/components/ui/card';
    27→import { ScrollArea } from '@/components/ui/scroll-area';
    28→import { Separator } from '@/components/ui/separator';
    29→import {
    30→  Breadcrumb,
    31→  BreadcrumbItem,
    32→  BreadcrumbLink,
    33→  BreadcrumbList,
    34→  BreadcrumbPage,
    35→  BreadcrumbSeparator,
    36→} from '@/components/ui/breadcrumb';
    37→import {
    38→  DropdownMenu,
    39→  DropdownMenuContent,
    40→  DropdownMenuItem,
    41→  DropdownMenuTrigger,
    42→} from '@/components/ui/dropdown-menu';
    43→import { UploadZone } from '@/components/upload/upload-zone';
    44→import { RenameDialog } from './rename-dialog';
    45→import { NoteEditor } from './note-editor';
    46→import { ErrorBoundary } from '@/components/error/error-boundary';
    47→import { NoteEditorError } from '@/components/error/note-editor-error';
    48→import { useDeleteNode } from '@/hooks/use-file-tree';
    49→import { buildTree } from '@/store/file-tree';
    50→import { toast } from 'sonner';
    51→import { DraggableItem } from '@/components/dnd/draggable-item';
    52→import { DroppableFolder } from '@/components/dnd/droppable-folder';
    53→import { useWorkspaceDnd } from '@/components/dnd/dnd-context';
    54→import { SearchDropdown } from '@/components/search/search-dropdown';
    55→import { ShareDialog } from '@/components/sharing/share-dialog';
    56→import { VersionListDialog } from '@/components/versions/version-list-dialog';
    57→import { RevisionSidebar } from '@/components/revisions/revision-sidebar';
    58→import { BulkActionToolbar } from '@/components/bulk/bulk-action-toolbar';
    59→
    60→export function ContentArea() {
    61→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid');
    62→  const [selectedNodeId, setSelectedNodeId] = useState<string | null>(null);
    63→  const [renameDialogOpen, setRenameDialogOpen] = useState(false);
    64→  const [renameNodeId, setRenameNodeId] = useState<string>('');
    65→  const [renameNodeName, setRenameNodeName] = useState<string>('');
    66→  const [multiSelectedIds, setMultiSelectedIds] = useState<Set<string>>(new Set());
    67→  const [shareDialogOpen, setShareDialogOpen] = useState(false);
    68→  const [shareNode, setShareNode] = useState<TreeNode | null>(null);
    69→  const [versionDialogOpen, setVersionDialogOpen] = useState(false);
    70→  const [versionNodeId, setVersionNodeId] = useState<string>('');
    71→  const [versionFileName, setVersionFileName] = useState<string>('');
    72→  const [showRevisionSidebar, setShowRevisionSidebar] = useState(false);
    73→
    74→  const {
    75→    tree,
    76→    flatNodes,
    77→    currentFolderId,
    78→    currentFolderPath,
    79→    setCurrentFolder,
    80→    isLoading,
    81→    selectedNodeIds,
    82→    selectNode,
    83→  } = useFileTreeStore();
    84→
    85→  const { isDragging } = useWorkspaceDnd();
    86→
    87→  const deleteMutation = useDeleteNode();
    88→
    89→  // 21 — Favorite toggle: We use a lazy approach to avoid creating a mutation per node
    90→  // since nodeId varies per item click
    91→  const handleFavoriteToggle = async (nodeId: string) => {
    92→    const node = flatNodes.get(nodeId);
    93→    if (!node) return;
    94→    const res = await fetch(`/api/nodes/${nodeId}/favorite`, {
    95→      method: 'PATCH',
    96→      headers: { 'Content-Type': 'application/json' },
    97→      body: JSON.stringify({ isFavorite: !node.isFavorite }),
    98→    });
    99→    const data = await res.json();
   100→    if (data.success) {
   101→      // Optimistic update: toggle isFavorite in the local tree
   102→      const flatNodesUpdate = new Map(flatNodes);
   103→      const existing = flatNodesUpdate.get(nodeId);
   104→      if (existing) {
   105→        flatNodesUpdate.set(nodeId, { ...existing, isFavorite: !existing.isFavorite });
   106→      }
   107→      // Force refresh from store
   108→      const storeNodes = Array.from(flatNodesUpdate.values());
   109→      const newTree = buildTree(storeNodes);
   110→      useFileTreeStore.getState().setTree(newTree);
   111→      toast.success(data.data.isFavorite ? 'Added to favorites' : 'Removed from favorites');
   112→    }
   113→  };
   114→
   115→  // Get items in the current folder
   116→  const currentFolder = flatNodes.get(currentFolderId || '');
   117→  const itemsInFolder = currentFolder?.children || tree;
   118→
   119→  // Search is now handled by SearchDropdown — items display is unfiltered locally
   120→  const filteredItems = itemsInFolder;
   121→
   122→  // Multi-select: get selected nodes for drag operations
   123→  const getSelectedNodes = (draggedNode: TreeNode): TreeNode[] => {
   124→    if (multiSelectedIds.size > 0 && multiSelectedIds.has(draggedNode.id)) {
   125→      return Array.from(multiSelectedIds)
   126→        .map(id => flatNodes.get(id))
   127→        .filter(Boolean) as TreeNode[];
   128→    }
   129→    return [draggedNode];
   130→  };
   131→
   132→  // Handle multi-select click (Ctrl/Cmd + click)
   133→  const handleItemClick = (node: TreeNode, e?: React.MouseEvent) => {
   134→    if (e && (e.metaKey || e.ctrlKey)) {
   135→      // Multi-select
   136→      const newSelection = new Set(multiSelectedIds);
   137→      if (newSelection.has(node.id)) {
   138→        newSelection.delete(node.id);
   139→      } else {
   140→        newSelection.add(node.id);
   141→      }
   142→      setMultiSelectedIds(newSelection);
   143→      return;
   144→    }
   145→
   146→    // Clear multi-selection on regular click
   147→    setMultiSelectedIds(new Set());
   148→
   149→    if (node.type === 'folder') {
   150→      navigateToFolder(node.id, node.name);
   151→    } else if (node.type === 'note') {
   152→      openNote(node.id);
   153→    } else {
   154→      setSelectedNodeId(node.id);
   155→    }
   156→  };
   157→
   158→  // Navigate to folder
   159→  const navigateToFolder = (folderId: string, folderName: string) => {
   160→    const newPath = [...currentFolderPath, { id: folderId, name: folderName }];
   161→    setCurrentFolder(folderId, newPath);
   162→    setSelectedNodeId(null);
   163→  };
   164→
   165→  // Navigate up
   166→  const navigateUp = () => {
   167→    if (currentFolderPath.length > 1) {
   168→      const parentPath = currentFolderPath.slice(0, -1);
   169→      const parentFolder = parentPath[parentPath.length - 1];
   170→      setCurrentFolder(parentFolder.id, parentPath);
   171→      setSelectedNodeId(null);
   172→    }
   173→  };
   174→
   175→  // Navigate via breadcrumb
   176→  const navigateBreadcrumb = (index: number) => {
   177→    const newPath = currentFolderPath.slice(0, index + 1);
   178→    const target = newPath[newPath.length - 1];
   179→    setCurrentFolder(target.id, newPath);
   180→    setSelectedNodeId(null);
   181→  };
   182→
   183→  // Open note editor
   184→  const openNote = (noteId: string) => {
   185→    setSelectedNodeId(noteId);
   186→  };
   187→
   188→  const handleRename = (nodeId: string, currentName: string) => {
   189→    setRenameNodeId(nodeId);
   190→    setRenameNodeName(currentName);
   191→    setRenameDialogOpen(true);
   192→  };
   193→
   194→  const handleDelete = (nodeId: string) => {
   195→    deleteMutation.mutate({ nodeId });
   196→  };
   197→
   198→  const handleShare = (node: TreeNode) => {
   199→    setShareNode(node);
   200→    setShareDialogOpen(true);
   201→  };
   202→
   203→  const handleVersionHistory = (node: TreeNode) => {
   204→    setVersionNodeId(node.id);
   205→    setVersionFileName(node.name);
   206→    setVersionDialogOpen(true);
   207→  };
   208→
   209→  const formatBytes = (bytes: number): string => {
   210→    if (!bytes || bytes < 0) return '—';
   211→    const k = 1024;
   212→    const sizes = ['B', 'KB', 'MB', 'GB'];
   213→    const i = Math.floor(Math.log(bytes) / Math.log(k));
   214→    return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`;
   215→  };
   216→
   217→  const formatDate = (dateStr: string): string => {
   218→    try {
   219→      const date = new Date(dateStr);
   220→      return date.toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' });
   221→    } catch {
   222→      return '—';
   223→    }
   224→  };
   225→
   226→  const getIcon = (type: NodeType) => {
   227→    switch (type) {
   228→      case 'folder':
   229→        return <Folder className="h-5 w-5 text-orange-500" />;
   230→      case 'file':
   231→        return <File className="h-5 w-5 text-muted-foreground" />;
   232→      case 'note':
   233→        return <FileText className="h-5 w-5 text-emerald-600" />;
   234→    }
   235→  };
   236→
   237→  // If a note is selected, show the note editor + optional revision sidebar
   238→  const selectedNode = selectedNodeId ? flatNodes.get(selectedNodeId) : null;
   239→  if (selectedNode && selectedNode.type === 'note') {
   240→    return (
   241→      <div className="flex h-full">
   242→        {/* Main content area */}
   243→        <div className={showRevisionSidebar ? 'flex-1 flex flex-col p-6 transition-all min-w-0' : 'flex-1 flex flex-col p-6 transition-all'}>
   244→          <div className="flex items-center gap-2 mb-4">
   245→            <Button
   246→              variant="ghost"
   247→              size="sm"
   248→              className="min-h-[44px]"
   249→              aria-label="Back to folder view"
   250→              onClick={() => {
   251→                setSelectedNodeId(null);
   252→                setShowRevisionSidebar(false);
   253→              }}
   254→            >
   255→              <ArrowUp className="h-4 w-4 mr-1" />
   256→              Back
   257→            </Button>
   258→            <Separator orientation="vertical" className="h-4" />
   259→            <h2 className="text-lg font-semibold">{selectedNode.name}</h2>
   260→            <Separator orientation="vertical" className="h-4" />
   261→            <Button
   262→              variant={showRevisionSidebar ? 'secondary' : 'ghost'}
   263→              size="sm"
   264→              className="min-h-[44px]"
   265→              aria-label="Toggle version history sidebar"
   266→              aria-expanded={showRevisionSidebar}
   267→              onClick={() => setShowRevisionSidebar(!showRevisionSidebar)}
   268→            >
   269→              <History className="h-4 w-4 mr-1" />
   270→              Version History
   271→            </Button>
   272→          </div>
   273→          <ErrorBoundary fallback={NoteEditorError} context={{ componentName: 'NoteEditor', action: 'render_note' }}>
   274→            <NoteEditor nodeId={selectedNode.id} />
   275→          </ErrorBoundary>
   276→        </div>
   277→
   278→        {/* Revision sidebar panel */}
   279→        <AnimatePresence>
   280→          {showRevisionSidebar && (
   281→            <motion.div
   282→              initial={{ width: 0, opacity: 0 }}
   283→              animate={{ width: 320, opacity: 1 }}
   284→              exit={{ width: 0, opacity: 0 }}
   285→              transition={{ duration: 0.2 }}
   286→              className="shrink-0 overflow-hidden"
   287→            >
   288→              <RevisionSidebar
   289→                nodeId={selectedNode.id}
   290→                onClose={() => setShowRevisionSidebar(false)}
   291→              />
   292→            </motion.div>
   293→          )}
   294→        </AnimatePresence>
   295→      </div>
   296→    );
   297→  }
   298→
   299→  return (
   300→    <div className="flex flex-col h-full">
   301→      {/* Toolbar */}
   302→      <div className="sticky top-0 z-10 bg-background/95 backdrop-blur border-b border-border px-6 py-3">
   303→        <div className="flex items-center gap-4">
   304→          {/* Breadcrumb — 29: wrapped in <nav> */}
   305→          <nav aria-label="Breadcrumb">
   306→          <Breadcrumb className="flex-1 min-w-0">
   307→            <BreadcrumbList>
   308→              {currentFolderPath.map((segment, index) => (
   309→                <Fragment key={`bc-${index}`}>
   310→                  {index > 0 && (
   311→                    <BreadcrumbSeparator>
   312→                      <ChevronRight className="h-3 w-3" />
   313→                    </BreadcrumbSeparator>
   314→                  )}
   315→                  <BreadcrumbItem>
   316→                    {index === currentFolderPath.length - 1 ? (
   317→                      <BreadcrumbPage className="text-sm truncate">
   318→                        {segment.name}
   319→                      </BreadcrumbPage>
   320→                    ) : (
   321→                      <BreadcrumbLink
   322→                        onClick={() => navigateBreadcrumb(index)}
   323→                        className="text-sm cursor-pointer truncate"
   324→                      >
   325→                        {segment.name}
   326→                      </BreadcrumbLink>
   327→                    )}
   328→                  </BreadcrumbItem>
   329→                </Fragment>
   330→              ))}
   331→            </BreadcrumbList>
   332→          </Breadcrumb>
   333→          </nav>
   334→
   335→
   336→
   337→          {/* Search — 12.4: debounced dropdown */}
   338→          <SearchDropdown
   339→            className="hidden sm:block w-48"
   340→            onNavigateToNode={(nodeId, nodeType, parentId) => {
   341→              setMultiSelectedIds(new Set());
   342→              if (nodeType === 'folder') {
   343→                const folderNode = flatNodes.get(nodeId);
   344→                if (folderNode) {
   345→                  navigateToFolder(nodeId, folderNode.name);
   346→                }
   347→              } else if (nodeType === 'note') {
   348→                // Navigate to parent folder then open note
   349→                if (parentId) {
   350→                  const parentFolder = flatNodes.get(parentId);
   351→                  if (parentFolder) {
   352→                    setCurrentFolder(parentId, []);
   353→                  }
   354→                }
   355→                openNote(nodeId);
   356→              } else {
   357→                // File — navigate to parent folder
   358→                if (parentId) {
   359→                  const parentFolder = flatNodes.get(parentId);
   360→                  if (parentFolder) {
   361→                    setCurrentFolder(parentId, []);
   362→                  }
   363→                }
   364→                setSelectedNodeId(nodeId);
   365→              }
   366→            }}
   367→          />
   368→
   369→          {/* View mode toggle */}
   370→          <div className="flex items-center border rounded-md">
   371→            <Button
   372→              variant={viewMode === 'grid' ? 'secondary' : 'ghost'}
   373→              size="icon"
   374→              className="h-8 w-8 min-h-[44px] min-w-[44px]"
   375→              onClick={() => setViewMode('grid')}
   376→              aria-label="Grid view"
   377→            >
   378→              <Grid3X3 className="h-4 w-4" />
   379→            </Button>
   380→            <Button
   381→              variant={viewMode === 'list' ? 'secondary' : 'ghost'}
   382→              size="icon"
   383→              className="h-8 w-8 min-h-[44px] min-w-[44px]"
   384→              onClick={() => setViewMode('list')}
   385→              aria-label="List view"
   386→            >
   387→              <List className="h-4 w-4" />
   388→            </Button>
   389→          </div>
   390→        </div>
   391→      </div>
   392→
   393→      {/* Bulk Action Toolbar — 18.2: appears when multi-select is active */}
   394→      {multiSelectedIds.size > 0 && (
   395→        <BulkActionToolbar
   396→          selectedIds={multiSelectedIds}
   397→          onClearSelection={() => setMultiSelectedIds(new Set())}
   398→        />
   399→      )}
   400→
   401→      {/* Content */}
   402→      <ScrollArea className="flex-1">
   403→        <div className="p-6">
   404→          {/* Upload Zone */}
   405→          <UploadZone />
   406→
   407→          {/* Loading state */}
   408→          {isLoading && (
   409→            <div className="flex items-center justify-center py-12">
   410→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   411→            </div>
   412→          )}
   413→
   414→          {/* Empty state */}
   415→          {!isLoading && filteredItems.length === 0 && (
   416→            <motion.div
   417→              initial={{ opacity: 0 }}
   418→              animate={{ opacity: 1 }}
   419→              className="text-center py-16"
   420→            >
   421→              <div className="inline-flex items-center justify-center w-16 h-16 rounded-full bg-muted mb-4">
   422→                <Folder className="h-8 w-8 text-muted-foreground" />
   423→              </div>
   424→              <h3 className="text-lg font-medium">This folder is empty</h3>
   425→              <p className="text-sm text-muted-foreground mt-1">
   426→                Drag files here or create a new folder/note
   427→              </p>
   428→            </motion.div>
   429→          )}
   430→
   431→          {/* Items */}
   432→          {!isLoading && filteredItems.length > 0 && (
   433→            <AnimatePresence mode="wait">
   434→              <motion.div
   435→                key={viewMode}
   436→                initial={{ opacity: 0 }}
   437→                animate={{ opacity: 1 }}
   438→                exit={{ opacity: 0 }}
   439→                transition={{ duration: 0.15 }}
   440→              >
   441→                {viewMode === 'grid' ? (
   442→                  <ul role="list" aria-label="Folder contents grid" className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
   443→                    {filteredItems.map((node) => {
   444→                      // Folder items get DroppableFolder wrapper + DraggableItem
   445→                      // Non-folder items just get DraggableItem
   446→                      const isSelected = multiSelectedIds.has(node.id);
   447→
   448→                      const cardContent = (
   449→                        <Card
   450→                          role="listitem"
   451→                          className={`cursor-pointer hover:border-accent transition-colors group relative
   452→                            ${isSelected ? 'ring-2 ring-emerald-500/50 bg-emerald-50/30 dark:bg-emerald-950/10' : ''}
   453→                            ${isDragging ? 'pointer-events-none' : ''}
   454→                          `}
   455→                          onClick={(e) => handleItemClick(node, e)}
   456→                        >
   457→                          <CardContent className="p-4 flex flex-col items-center text-center gap-2">
   458→                            <div className="w-12 h-12 rounded-lg bg-muted flex items-center justify-center">
   459→                              {getIcon(node.type)}
   460→                            </div>
   461→                            <span className="text-sm font-medium truncate max-w-full">
   462→                              {node.name}
   463→                            </span>
   464→                            <div className="flex items-center gap-1 text-xs text-muted-foreground">
   465→                              {node.type === 'folder' && (
   466→                                <span>{node.children?.length || 0} items</span>
   467→                              )}
   468→                              {node.type === 'file' && node.metadata && (
   469→                                <span>{formatBytes(node.metadata.sizeBytes)}</span>
   470→                              )}
   471→                              {node.type === 'note' && (
   472→                                <span>Note</span>
   473→                              )}
   474→                            </div>
   475→                            <span className="text-xs text-muted-foreground">
   476→                              {formatDate(node.updatedAt)}
   477→                            </span>
   478→                          </CardContent>
   479→
   480→                          {/* Actions overlay */}
   481→                          <div className="absolute top-2 right-2 opacity-0 group-hover:opacity-100 transition-opacity">
   482→                            <DropdownMenu>
   483→                              <DropdownMenuTrigger asChild>
   484→                                <Button variant="ghost" size="icon" className="h-7 w-7 min-h-[44px] min-w-[44px]" aria-label={`More actions for ${node.name}`}>
   485→                                  <MoreHorizontal className="h-3.5 w-3.5" />
   486→                                </Button>
   487→                              </DropdownMenuTrigger>
   488→                              <DropdownMenuContent align="end">
   489→                                <DropdownMenuItem onClick={() => handleFavoriteToggle(node.id)}>
   490→                                  <Star className={`h-4 w-4 mr-2 ${node.isFavorite ? 'fill-yellow-400 text-yellow-400' : ''}`} />
   491→                                  {node.isFavorite ? 'Remove from Favorites' : 'Add to Favorites'}
   492→                                </DropdownMenuItem>
   493→                                <DropdownMenuItem onClick={() => handleRename(node.id, node.name)}>
   494→                                  <Pencil className="h-4 w-4 mr-2" />
   495→                                  Rename
   496→                                </DropdownMenuItem>
   497→                                <DropdownMenuItem onClick={() => handleShare(node)}>
   498→                                  <Share2 className="h-4 w-4 mr-2" />
   499→                                  Share
   500→                                </DropdownMenuItem>
   501→                                {node.type === 'file' && (
   502→                                  <DropdownMenuItem onClick={() => handleVersionHistory(node)}>
   503→                                    <History className="h-4 w-4 mr-2" />
   504→                                    Version History
   505→                                  </DropdownMenuItem>
   506→                                )}
   507→                                <DropdownMenuItem
   508→                                  onClick={() => handleDelete(node.id)}
   509→                                  className="text-destructive focus:text-destructive"
   510→                                >
   511→                                  <Trash2 className="h-4 w-4 mr-2" />
   512→                                  Delete
   513→                                </DropdownMenuItem>
   514→                              </DropdownMenuContent>
   515→                            </DropdownMenu>
   516→                          </div>
   517→                        </Card>
   518→                      );
   519→
   520→                      // Wrap folder cards with DroppableFolder for drop targets
   521→                      if (node.type === 'folder') {
   522→                        return (
   523→                          <DraggableItem
   524→                            key={node.id}
   525→                            id={node.id}
   526→                            node={node}
   527→                            selectedNodes={getSelectedNodes(node)}
   528→                          >
   529→                            <DroppableFolder id={node.id} node={node}>
   530→                              {cardContent}
   531→                            </DroppableFolder>
   532→                          </DraggableItem>
   533→                        );
   534→                      }
   535→
   536→                      return (
   537→                        <DraggableItem
   538→                          key={node.id}
   539→                          id={node.id}
   540→                          node={node}
   541→                          selectedNodes={getSelectedNodes(node)}
   542→                        >
   543→                          {cardContent}
   544→                        </DraggableItem>
   545→                      );
   546→                    })}
   547→                  </ul>
   548→                ) : (
   549→                  <ul role="list" aria-label="Folder contents list" className="space-y-1">
   550→                    {filteredItems.map((node) => {
   551→                      const isSelected = multiSelectedIds.has(node.id);
   552→
   553→                      const rowContent = (
   554→                        <li
   555→                          role="listitem"
   556→                          className={`flex items-center gap-3 p-3 rounded-md hover:bg-accent/50 cursor-pointer group transition-colors
   557→                            ${isSelected ? 'ring-2 ring-emerald-500/50 bg-emerald-50/30 dark:bg-emerald-950/10' : ''}
   558→                            ${isDragging ? 'pointer-events-none' : ''}
   559→                          `}
   560→                          onClick={(e) => handleItemClick(node, e)}
   561→                        >
   562→                          <div className="w-8 h-8 rounded flex items-center justify-center shrink-0">
   563→                            {getIcon(node.type)}
