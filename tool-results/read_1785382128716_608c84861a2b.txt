    94→  const { data: onboardingData, isLoading: onboardingLoading } = useQuery({

    98→      const data = await res.json();
    99→      if (!data.success) {
   100→        // Return default onboarding state instead of undefined
   101→        return {
   102→          welcomeCompleted: false,
   103→          sampleContentLoaded: false,
   104→          checklistProgress: {},
   105→          dismissedAt: null,
   106→          steps: [],
   107→        };
   108→      }
   109→      return data.data as {
   110→        welcomeCompleted: boolean;
   111→        sampleContentLoaded: boolean;
   112→        checklistProgress: Record<string, boolean>;

   119→

   122→

   186→    if (!target) return false;
   187→    const tagName = target.tagName.toLowerCase();
   188→    // Check if target is input, textarea, or contenteditable
   189→    if (tagName === 'input' || tagName === 'textarea') return true;
   190→    if (target.isContentEditable) return true;

   192→    if (target.closest('[cmdk-input]')) return true;
   193→    return false;

   195→
   196→  // 22 — Global keyboard shortcuts (single useEffect with keydown listener)
   197→  const handleKeyDown = useCallback(
   198→    (e: KeyboardEvent) => {
   199→      // Command Palette: Cmd/Ctrl+K (new shortcut, replaces old calculator shortcut)
   200→      if ((e.metaKey || e.ctrlKey) && !e.shiftKey && e.key === 'k') {
   201→        e.preventDefault();
   202→        setCommandPaletteOpen(true);
   203→        markOnboardingStep('use_command_palette');
   204→        return;

   208→      if ((e.metaKey || e.ctrlKey) && e.shiftKey && (e.key === 'K' || e.key === 'k')) {
   209→        e.preventDefault();
   210→        toggleOpen();
   211→        markOnboardingStep('use_calculator');
   212→        return;
   213→      }
   214→
   215→      // Search: Cmd/Ctrl+Shift+F
   216→      if ((e.ctrlKey || e.metaKey) && e.shiftKey && (e.key === 'F' || e.key === 'f')) {
   217→        e.preventDefault();
   218→        setHeaderSearchOpen(true);
   219→        markOnboardingStep('use_search');
   220→        setTimeout(() => {
   221→          const input = searchRef.current?.querySelector('input');
   222→          if (input) input.focus();
   223→        }, 50);
   224→        return;
   225→      }
   226→
   227→      // Undo: Cmd/Ctrl+Z
   228→      if ((e.metaKey || e.ctrlKey) && e.key === 'z' && !e.shiftKey) {
   229→        e.preventDefault();
   230→        const action = popAction();
   231→        if (action) {
   232→          // Note: actual undo logic would be implemented per action type
   233→          // This just pops from the stack; the real undo needs to be wired per action
   234→          console.log('Undo action:', action.description);
   235→        }
   236→        return;
   237→      }
   238→
   239→      // Single-key shortcuts only work when user is NOT typing
   240→      if (isUserTyping(e)) return;
   241→
   242→      // Only process single-key shortcuts when no dialog/palette is open
   243→      if (commandPaletteOpen) return;
   244→
   245→      // New Note: N key
   246→      if (e.key === 'n' || e.key === 'N') {
   247→        e.preventDefault();
   248→        setPaletteCreateType('note');
   249→        setPaletteCreateDialogOpen(true);
   250→        return;
   251→      }
   252→
   253→      // New Folder: F key
   254→      if (e.key === 'f' || e.key === 'F') {
   255→        e.preventDefault();
   256→        setPaletteCreateType('folder');
   257→        setPaletteCreateDialogOpen(true);
   258→        return;
   259→      }
   260→
   261→      // Delete selected: Delete/Backspace key
   262→      if (e.key === 'Delete' || e.key === 'Backspace') {
   263→        const selectedIds = Array.from(selectedNodeIds);
   264→        if (selectedIds.length > 0) {
   265→          e.preventDefault();
   266→          // Delete the first selected item
   267→          deleteMutation.mutate({ nodeId: selectedIds[0] });
   268→        }
   269→        return;
   270→      }
   271→    },
   272→    [toggleOpen, isUserTyping, popAction, commandPaletteOpen, selectedNodeIds, deleteMutation]
   273→  );
   274→
   275→  useEffect(() => {

   278→  }, [handleKeyDown]);

   288→      const folderNode = flatNodes.get(nodeId);
   289→      if (folderNode) {
   290→        setCurrentFolder(nodeId, []);
   291→      }
   292→    } else if (nodeType === 'note') {
   293→      if (parentId) {
   294→        setCurrentFolder(parentId, []);
   295→      }
   296→    } else {
   297→      if (parentId) {
   298→        setCurrentFolder(parentId, []);
   299→      }
   300→    }
   301→  }, [flatNodes, setCurrentFolder]);
   302→

   304→  const toggleSidebar = useCallback(() => {
   305→    if (isMobile) {
   306→      setSidebarOpen(prev => !prev);

   308→      if (sidebarOpen) {
   309→        setSidebarCollapsed(!sidebarCollapsed);
   310→      } else {
   311→        setSidebarOpen(true);
   312→        setSidebarCollapsed(false);
   313→      }
   314→    }

   319→    setSidebarOpen(false);
   320→  }, []);
   321→

   324→  const sheetOpacity = useTransform(sheetY, [0, 200], [1, 0]);
   325→

   327→  const handleSheetPan = useCallback((event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {

   329→      closeSidebar();
   330→    }
   331→  }, [closeSidebar]);
   332→
   333→
   334→
   335→  return (
   336→    <WorkspaceDndProvider>
   337→      <TooltipProvider>

   339→          {/* 29 — Skip-to-content link for keyboard users */}
   340→          <a
   341→            href="#main-content"
   342→            className="sr-only focus:not-sr-only focus:fixed focus:top-2 focus:left-2 focus:z-[100] focus:bg-primary focus:text-primary-foreground focus:px-4 focus:py-2 focus:rounded-md focus:text-sm focus:font-medium focus:shadow-lg"
   343→          >
   344→            Skip to main content
   345→          </a>

   401→                    aria-label="Open command palette"
   402→                  >
   403→                    <Search className="h-5 w-5" />
   404→                  </Button>
   405→                )}
   406→              </div>

   408→              <div className="flex items-center gap-3">
   409→                {/* MODUL 47 — Locale switcher */}
   410→                <LocaleSwitcher />
   411→
   412→                {/* Notification badge */}
   413→                <NotificationBadge />
   414→
   415→                {/* Calculator toggle button — updated shortcut tooltip */}
   416→                <Tooltip>
   417→                  <TooltipTrigger asChild>
   418→                    <Button
   419→                      variant={isOpen ? 'secondary' : 'ghost'}
   420→                      size="icon"
   421→                      className="h-9 w-9 shrink-0 min-h-[44px] min-w-[44px]"
   422→                      onClick={toggleOpen}
   423→                      aria-label="Toggle calculator (Ctrl+Shift+K)"
   424→                    >
   425→                      <Calculator className={`h-4 w-4 ${isOpen ? 'text-orange-500' : 'text-muted-foreground'}`} />

   433→                {/* User menu dropdown */}

   437→                      <div className="flex items-center justify-center w-8 h-8 rounded-full bg-neutral-200 dark:bg-neutral-800 text-neutral-700 dark:text-neutral-300 text-sm font-medium">

   441→                    </Button>

   443→                  <DropdownMenuContent align="end" className="w-56">

   445→                      <div className="flex flex-col">
   446→                        <span className="font-medium">{user?.name || 'User'}</span>
   447→                        <span className="text-xs text-muted-foreground">{user?.email}</span>
   448→                      </div>
   449→                    </DropdownMenuLabel>
   450→                    <DropdownMenuSeparator />
   451→                    {currentWorkspaceId && (currentWorkspaceRole === 'owner' || currentWorkspaceRole === 'admin') && (

   453→                        onClick={() => setWorkspaceSettingsOpen(true)}
   454→                      >
   455→                        <Building2 className="h-4 w-4 mr-2" />
   456→                        Workspace Settings
   457→                      </DropdownMenuItem>
   458→                    )}

   460→                      onClick={() => setInvitationViewOpen(true)}
   461→                    >
   462→                      <Mail className="h-4 w-4 mr-2" />
   463→                      Invitations
   464→                    </DropdownMenuItem>
   465→                    <DropdownMenuItem
   466→                      onClick={() => setSettingsDialogOpen(true)}
   467→                    >
   468→                      <Settings className="h-4 w-4 mr-2" />
   469→                      Settings
   470→                    </DropdownMenuItem>

   476→                      }}
   477→                      className="text-destructive focus:text-destructive"
   478→                    >
   479→                      <LogOut className="h-4 w-4 mr-2" />
   480→                      Sign Out
   481→                    </DropdownMenuItem>
   482→                  </DropdownMenuContent>
   483→                </DropdownMenu>
   484→              </div>
   485→            </div>
   486→          </header>
   487→
   488→          {/* Main content: sidebar + workspace */}
   489→          <div className="flex-1 flex overflow-hidden relative">
   490→            {/* 29 — aria-live region for toast announcements */}
   491→            <div aria-live="polite" aria-atomic="true" className="sr-only" id="a11y-announcements" />
   492→
   493→            {/* ===== Mobile: Bottom-sheet drawer pattern ===== */}
   494→            {isMobile && (
   495→              <AnimatePresence initial={false}>
   496→                {sidebarOpen && (
   497→                  <>
   498→                    {/* Backdrop */}
   499→                    <motion.div
   500→                      initial={{ opacity: 0 }}
   501→                      animate={{ opacity: 1 }}
   502→                      exit={{ opacity: 0 }}
   503→                      transition={{ duration: 0.2 }}
   504→                      className="fixed inset-0 bg-black/40 z-20"
   505→                      onClick={closeSidebar}
   506→                      role="button"
   507→                      tabIndex={-1}
   508→                      aria-label="Close sidebar overlay"
   509→                    />
   510→
   511→                    {/* Bottom sheet */}
   512→                    <motion.aside
   513→                      role="complementary"
   514→                      aria-label="Sidebar navigation"
   515→                      initial={{ y: '100%' }}
   516→                      animate={{ y: 0 }}
   517→                      exit={{ y: '100%' }}
   518→                      transition={{ type: 'spring', damping: 30, stiffness: 300 }}
   519→                      drag="y"
   520→                      dragConstraints={{ top: 0 }}
   521→                      dragElastic={0.2}
   522→                      onDragEnd={handleSheetPan}
   523→                      style={{ y: sheetY, opacity: sheetOpacity }}
   524→                      className="fixed bottom-0 left-0 right-0 z-20 bg-sidebar border-t border-border rounded-t-2xl overflow-hidden shadow-xl"
   525→                    >
   526→                      {/* Drag handle bar */}
   527→                      <div className="flex items-center justify-center pt-3 pb-2 cursor-grab active:cursor-grabbing">
