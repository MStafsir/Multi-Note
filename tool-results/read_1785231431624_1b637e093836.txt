    77→    77→// Renders xlsx data as a scrollable table with sheet tabs

    79→    79→function SpreadsheetPreview({ data, name }: {
    80→    80→  data: {
    81→    81→    sheetNames: string[];
    82→    82→    sheets: Record<string, { rows: Record<string, string | number | boolean | null>[]; headers: string[] }>;
    83→    83→  };
    84→    84→  name: string;
    85→    85→}) {
    86→    86→  const [activeSheet, setActiveSheet] = useState(data.sheetNames[0] || '');
    87→    87→  const sheetData = data.sheets[activeSheet];
    88→    88→  const maxRows = 100;
    89→    89→
    90→    90→  if (!sheetData) {
    91→    91→    return <p className="text-sm text-muted-foreground">No data found in spreadsheet</p>;
    92→    92→  }
    93→    93→
    94→    94→  return (
    95→    95→    <div className="flex flex-col w-full">
    96→    96→      {/* Sheet tabs */}
    97→    97→      {data.sheetNames.length > 1 && (
    98→    98→        <div className="flex items-center gap-1 mb-3 overflow-x-auto pb-1">
    99→    99→          {data.sheetNames.map((sheetName) => (
   100→   100→            <Button
   101→   101→              key={sheetName}
   102→   102→              variant={sheetName === activeSheet ? 'secondary' : 'ghost'}
   103→   103→              size="sm"
   104→   104→              className="text-xs whitespace-nowrap"
   105→   105→              onClick={() => setActiveSheet(sheetName)}
   106→   106→            >
   107→   107→              {sheetName}
   108→   108→            </Button>
   109→   109→          ))}
   110→   110→        </div>
   111→   111→      )}
   112→   112→
   113→   113→      {/* Table */}
   114→   114→      <div className="overflow-auto max-h-[60vh] border rounded-lg">
   115→   115→        <table className="w-full text-sm">
   116→   116→          <thead className="sticky top-0 bg-muted/80 backdrop-blur">
   117→   117→            <tr>
   118→   118→              {sheetData.headers.map((header, i) => (
   119→   119→                <th key={i} className="px-3 py-2 text-left font-medium border-b whitespace-nowrap">
   120→   120→                  {header || `Column ${i + 1}`}
   121→   121→                </th>
   122→   122→              ))}
   123→   123→            </tr>
   124→   124→          </thead>
   125→   125→          <tbody>
   126→   126→            {sheetData.rows.slice(0, maxRows).map((row, i) => (
   127→   127→              <tr key={i} className="hover:bg-accent/30 transition-colors">
   128→   128→                {sheetData.headers.map((header, j) => (
   129→   129→                  <td key={j} className="px-3 py-1.5 border-b whitespace-nowrap max-w-[200px] truncate">
   130→   130→                    {row[header] !== null && row[header] !== undefined ? String(row[header]) : ''}
   131→   131→                  </td>
   132→   132→                ))}
   133→   133→              </tr>
   134→   134→            ))}
   135→   135→          </tbody>
   136→   136→        </table>
   137→   137→        {sheetData.rows.length > maxRows && (
   138→   138→          <div className="p-2 text-center text-xs text-muted-foreground bg-muted/30">
   139→   139→            Showing {maxRows} of {sheetData.rows.length} rows
   140→   140→          </div>
   141→   141→        )}
   142→   142→        {sheetData.rows.length === 0 && (
   143→   143→          <div className="p-6 text-center text-sm text-muted-foreground">
   144→   144→            This sheet is empty
   145→   145→          </div>
   146→   146→        )}
   147→   147→      </div>
   148→   148→    </div>
   149→   149→  );
   150→   150→}

   219→   219→function PdfPreview({ contentUrl, name, sizeBytes, mimeLabel, closeButton }: {
   220→   220→  contentUrl: string;
   221→   221→  name: string;
   222→   222→  sizeBytes?: number;
   223→   223→  mimeLabel: string;
   224→   224→  closeButton: React.ReactNode;
   225→   225→}) {
   226→   226→  const [loading, setLoading] = useState(true);
   227→   227→  const [error, setError] = useState(false);
   228→   228→  const [numPages, setNumPages] = useState(0);
   229→   229→  const [currentPage, setCurrentPage] = useState(1);
   230→   230→  const [scale, setScale] = useState(1.2);
   231→   231→  const canvasRefs = useRef<Map<number, HTMLCanvasElement>>(new Map());
   232→   232→  const containerRef = useRef<HTMLDivElement>(null);
   233→   233→
   234→   234→  const renderPage = useCallback(async (pageNum: number, pdfDoc: unknown, scale: number) => {
   235→   235→    const pdfDocument = pdfDoc as { getPage: (n: number) => Promise<{ render: (params: unknown) => Promise<void>; viewport: { width: number; height: number; scale: number } }> };
   236→   236→    const page = await pdfDocument.getPage(pageNum);
   237→   237→    const viewport = page.getViewport({ scale });
   238→   238→
   239→   239→    let canvas = canvasRefs.current.get(pageNum);
   240→   240→    if (!canvas) {
   241→   241→      canvas = document.createElement('canvas');
   242→   242→      canvasRefs.current.set(pageNum, canvas);
   243→   243→      if (containerRef.current) {
   244→   244→        containerRef.current.appendChild(canvas);
   245→   245→      }
   246→   246→    }
   247→   247→
   248→   248→    canvas.width = viewport.width;
   249→   249→    canvas.height = viewport.height;
   250→   250→    canvas.style.width = `${viewport.width}px`;
   251→   251→    canvas.style.height = `${viewport.height}px`;
   252→   252→    canvas.style.marginBottom = '8px';
   253→   253→    canvas.style.display = 'block';
   254→   254→
   255→   255→    const context = canvas.getContext('2d');
   256→   256→    if (!context) return;
   257→   257→
   258→   258→    await page.render({
   259→   259→      canvasContext: context,
   260→   260→      viewport,
   261→   261→    });
   262→   262→  }, []);
   263→   263→
   264→   264→  useEffect(() => {
   265→   265→    let cancelled = false;
   266→   266→
   267→   267→    const loadPdf = async () => {
   268→   268→      try {
   269→   269→        // Dynamic import of pdfjs-dist
   270→   270→        const pdfjsLib = await import('pdfjs-dist');
   271→   271→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   272→   272→
   273→   273→        // Fetch PDF as ArrayBuffer from contentUrl
   274→   274→        const res = await fetch(contentUrl);
   275→   275→        if (!res.ok) throw new Error('Failed to fetch PDF');
   276→   276→        const arrayBuffer = await res.arrayBuffer();
   277→   277→
   278→   278→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   279→   279→        if (!cancelled) {
   280→   280→          setNumPages(pdfDoc.numPages);
   281→   281→          setLoading(false);
   282→   282→        }
   283→   283→
   284→   284→        // Render all pages
   285→   285→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   286→   286→          if (cancelled) break;
   287→   287→          await renderPage(i, pdfDoc, scale);
   288→   288→        }
   289→   289→      } catch (err) {
   290→   290→        if (!cancelled) {
   291→   291→          console.error('PDF preview error:', err);
   292→   292→          setError(true);
   293→   293→          setLoading(false);
   294→   294→        }
   295→   295→      }
   296→   296→    };
   297→   297→
   298→   298→    loadPdf();
   299→   299→    return () => { cancelled = true; };
   300→   300→  }, [contentUrl, renderPage]);
   301→   301→
   302→   302→  // Re-render pages when scale changes
   303→   303→  useEffect(() => {
   304→   304→    if (numPages === 0 || loading) return;
   305→   305→    let cancelled = false;
   306→   306→
   307→   307→    const rerender = async () => {
   308→   308→      try {
   309→   309→        const pdfjsLib = await import('pdfjs-dist');
   310→   310→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjsLib.version}/build/pdf.worker.min.mjs`;
   311→   311→
   312→   312→        const res = await fetch(contentUrl);
   313→   313→        if (!res.ok) return;
   314→   314→        const arrayBuffer = await res.arrayBuffer();
   315→   315→        const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
   316→   316→
   317→   317→        // Clear existing canvases
   318→   318→        if (containerRef.current) {
   319→   319→          containerRef.current.innerHTML = '';
   320→   320→        }
   321→   321→        canvasRefs.current.clear();
   322→   322→
   323→   323→        for (let i = 1; i <= pdfDoc.numPages; i++) {
   324→   324→          if (cancelled) break;
   325→   325→          await renderPage(i, pdfDoc, scale);
   326→   326→        }
   327→   327→      } catch {
   328→   328→        // Silently fail on re-render
   329→   329→      }
   330→   330→    };
   331→   331→
   332→   332→    rerender();
   333→   333→    return () => { cancelled = true; };
   334→   334→  }, [scale, numPages, contentUrl, renderPage]);
   335→   335→
   336→   336→  if (error) {
   337→   337→    return (
   338→   338→      <div className="relative flex flex-col w-full">
   339→   339→        {closeButton}
   340→   340→        <Card className="w-full">
   341→   341→          <CardContent className="p-6 flex flex-col items-center gap-4">
   342→   342→            <FileText className="h-12 w-12 text-muted-foreground" />
   343→   343→            <p className="text-sm text-muted-foreground">Failed to load PDF preview</p>
   344→   344→            <Button variant="outline" size="sm" asChild>
   345→   345→              <a href={contentUrl} target="_blank" rel="noopener noreferrer">
   346→   346→                <ExternalLink className="h-4 w-4 mr-2" />
   347→   347→                Open PDF directly
   348→   348→              </a>
   349→   349→            </Button>
   350→   350→          </CardContent>
   351→   351→        </Card>
   352→   352→      </div>
   353→   353→    );
   354→   354→  }
   355→   355→
   356→   356→  return (
   357→   357→    <div className="relative flex flex-col w-full">
   358→   358→      {closeButton}
   359→   359→      <div className="flex items-center gap-2 mb-3">
   360→   360→        <FileText className="h-5 w-5 text-red-500" />
   361→   361→        <span className="font-medium truncate max-w-xs">{name}</span>
   362→   362→        {sizeBytes && (
   363→   363→          <span className="text-sm text-muted-foreground">{formatFileSize(sizeBytes)}</span>
   364→   364→        )}
   365→   365→        <span className="text-xs text-muted-foreground">{mimeLabel}</span>
   366→   366→        <OfflineBadge />
   367→   367→        <div className="flex items-center gap-1 ml-auto">
   368→   368→          {numPages > 1 && (
   369→   369→            <>
   370→   370→              <Button
   371→   371→                variant="outline"
   372→   372→                size="icon"
   373→   373→                className="h-7 w-7"
   374→   374→                onClick={() => setCurrentPage(Math.max(1, currentPage - 1))}
   375→   375→                disabled={currentPage === 1}
   376→   376→                aria-label="Previous page"
   377→   377→              >
   378→   378→                <ChevronLeft className="h-3 w-3" />
   379→   379→              </Button>
   380→   380→              <span className="text-xs text-muted-foreground">
   381→   381→                Page {currentPage}/{numPages}
   382→   382→              </span>
   383→   383→              <Button
   384→   384→                variant="outline"
   385→   385→                size="icon"
   386→   386→                className="h-7 w-7"
   387→   387→                onClick={() => setCurrentPage(Math.min(numPages, currentPage + 1))}
   388→   388→                disabled={currentPage === numPages}
   389→   389→                aria-label="Next page"
   390→   390→              >
   391→   391→                <ChevronRight className="h-3 w-3" />
   392→   392→              </Button>
   393→   393→            </>
   394→   394→          )}
   395→   395→          <Button
   396→   396→            variant="outline"
   397→   397→            size="icon"
   398→   398→            className="h-7 w-7"
   399→   399→            onClick={() => setScale(Math.max(0.5, scale - 0.2))}
   400→   400→            disabled={scale <= 0.5}
   401→   401→            aria-label="Zoom out"
   402→   402→          >
   403→   403→            <ZoomOut className="h-3 w-3" />
   404→   404→          </Button>
   405→   405→          <span className="text-xs text-muted-foreground">{Math.round(scale * 100)}%</span>
   406→   406→          <Button
   407→   407→            variant="outline"
   408→   408→            size="icon"
   409→   409→            className="h-7 w-7"
   410→   410→            onClick={() => setScale(Math.min(3, scale + 0.2))}
   411→   411→            disabled={scale >= 3}
   412→   412→            aria-label="Zoom in"
   413→   413→          >
   414→   414→            <ZoomIn className="h-3 w-3" />
   415→   415→          </Button>
   416→   416→        </div>
   417→   417→      </div>
   418→   418→
   419→   419→      {loading ? (
   420→   420→        <div className="flex items-center justify-center py-12">
   421→   421→          <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   422→   422→          <span className="ml-2 text-sm text-muted-foreground">Loading PDF…</span>
   423→   423→        </div>
   424→   424→      ) : (
   425→   425→        <ScrollArea className="w-full rounded-lg border bg-gray-50 dark:bg-gray-900" style={{ maxHeight: '70vh' }}>
   426→   426→          <div
   427→   427→            ref={containerRef}
   428→   428→            className="flex flex-col items-center p-4"
   429→   429→          />
   430→   430→        </ScrollArea>
   431→   431→      )}
   432→   432→    </div>
   433→   433→  );
   434→   434→}

   436→   436→// ============================================================
   437→   437→// DOCX Preview (Tier 2 — docx-preview with mammoth fallback)
   438→   438→// ============================================================
   439→   439→function DocxPreview({ contentUrl, name, sizeBytes, mimeLabel, downloadUrl, closeButton }: {
   440→   440→  contentUrl: string;
   441→   441→  name: string;
   442→   442→  sizeBytes?: number;
   443→   443→  mimeLabel: string;
   444→   444→  downloadUrl: string;
   445→   445→  closeButton: React.ReactNode;
   446→   446→}) {
   447→   447→  const [loading, setLoading] = useState(true);
   448→   448→  const [error, setError] = useState<string | null>(null);
   449→   449→  const [mammothHtml, setMammothHtml] = useState<string | null>(null);
   450→   450→  const docxContainerRef = useRef<HTMLDivElement>(null);
   451→   451→
   452→   452→  useEffect(() => {
   453→   453→    let cancelled = false;
   454→   454→
   455→   455→    const loadDocx = async () => {
   456→   456→      try {
   457→   457→        // Fetch raw bytes from contentUrl
   458→   458→        const res = await fetch(contentUrl);
   459→   459→        if (!res.ok) throw new Error('Failed to fetch document');
   460→   460→        const arrayBuffer = await res.arrayBuffer();
   461→   461→
   462→   462→        // Try docx-preview first
   463→   463→        try {
   464→   464→          const docxPreview = await import('docx-preview');
   465→   465→          if (docxContainerRef.current && !cancelled) {
   466→   466→            await docxPreview.renderAsync(
   467→   467→              arrayBuffer,
   468→   468→              docxContainerRef.current,
   469→   469→              undefined,
   470→   470→              { className: 'docx-preview-wrapper' }
   471→   471→            );
   472→   472→            if (!cancelled) {
   473→   473→              setLoading(false);
   474→   474→            }
   475→   475→            return;
   476→   476→          }
   477→   477→        } catch (docxPreviewErr) {
   478→   478→          console.warn('docx-preview failed, falling back to mammoth:', docxPreviewErr);
   479→   479→        }
   480→   480→
   481→   481→        // Fallback to mammoth
   482→   482→        if (!cancelled) {
   483→   483→          const mammoth = await import('mammoth');
   484→   484→          const result = await mammoth.convertToHtml({ buffer: arrayBuffer });
   485→   485→          if (!cancelled) {
